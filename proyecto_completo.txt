WA_WS_URL = "wss://web.whatsapp.com/ws/chat"
WA_ORIGIN = "https://web.whatsapp.com"
WA_HEADER = b"WA\x06\x03"
WA_USER_AGENT = (
    "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
    "AppleWebKit/537.36 (KHTML, like Gecko) "
    "Chrome/120.0.0.0 Safari/537.36"
)
FALLBACK_WA_VERSION = (2, 3000, 1056109635)


class SnakeChatClient:

    def __init__(self, debug: bool = False, phone_number: str | None = None):
        self.debug = debug
        self.phone_number = phone_number
        self.wa_version = self._fetch_latest_wa_version()
        self.pairing = PairingState(self.wa_version)
        self.prekeys = PreKeyStore()
        self.signal: SignalSessionManager | None = None

        self._ws = None
        self._transport: NoiseTransport | None = None
        self._handlers: dict[str, list[Callable[..., Awaitable[Any]]]] = defaultdict(list)
        self._pending_iqs: dict[str, asyncio.Future] = {}
        self._media_conn: dict = {}
        self._prekey_seq: int = 1
        self._pairing_code_requested: bool = False

    def on(self, event: str):
        def decorator(func: Callable[..., Awaitable[Any]]):
            self._handlers[event].append(func)
            return func
        return decorator

    async def _run_handler(self, event: str, handler: Callable[..., Awaitable[Any]], *args):
        try:
            await handler(*args)
        except Exception as e:
            print(f"[!] Error en handler de '{event}': {type(e).__name__}: {e}")

    async def _emit(self, event: str, *args):
        for handler in self._handlers.get(event, []):
            asyncio.create_task(self._run_handler(event, handler, *args))

    @staticmethod
    def _fetch_latest_wa_version() -> tuple[int, int, int]:
        try:
            req = urllib.request.Request(
                "https://web.whatsapp.com/sw.js",
                headers={"User-Agent": WA_USER_AGENT},
            )
            with urllib.request.urlopen(req, timeout=8) as resp:
                text = resp.read().decode("utf-8", errors="ignore")
            match = re.search(r'"client_revision"\s*:\s*(\d+)', text)
            if match:
                return (2, 3000, int(match.group(1)))
        except Exception:
            pass
        return FALLBACK_WA_VERSION

    @staticmethod
    def _build_frame(payload: bytes, include_header: bool = False) -> bytes:
        length_prefix = struct.pack(">I", len(payload))[1:]
        return (WA_HEADER + length_prefix + payload) if include_header else (length_prefix + payload)

    @staticmethod
    def _unpack_transport(raw_decrypted: bytes) -> bytes:
        if not raw_decrypted:
            return b""
        flags = raw_decrypted[0]
        data = raw_decrypted[1:]
        return zlib.decompress(data) if (flags & 2) else data

    async def request_pairing_code(self, phone_number: str, custom_code: str | None = None) -> str:
        """
        Solicita un código de emparejamiento de 8 dígitos para vincular sin QR.
        Ejemplo: await client.request_pairing_code("50557418454")
        """
        code, iq_node = self.pairing.build_pairing_code_iq(phone_number, custom_code=custom_code)
        resp = await send_iq_and_wait(
            self._ws,
            self._transport,
            self._pending_iqs,
            iq_node,
            debug=self.debug,
        )
        if resp.attrs.get("type") == "error":
            raise RuntimeError(f"WhatsApp rechazó el Pairing Code: {resp}")

        self.pairing.print_pairing_code(code, self.pairing.pairing_phone or phone_number)
        await self._emit("pairing_code", code)
        return code

    async def _connect_once(self) -> bool:
        if self.pairing.is_registered:
            mode = "LOGIN"
        elif self.phone_number:
            mode = "CODE"
        else:
            mode = "QR"
        print(f"[*] Conectando a WhatsApp ({mode})...")

        should_reconnect = False
        keepalive_task: asyncio.Task | None = None
        self._pairing_code_requested = False

        self.signal = SignalSessionManager(
            ident_priv_bytes=self.pairing.ident_priv_bytes,
            ident_pub_bytes=self.pairing.ident_pub_bytes,
            skey_priv_bytes=self.pairing.skey_priv_bytes,
            prekeys=self.prekeys,
        )

        async with websockets.connect(
            WA_WS_URL,
            origin=WA_ORIGIN,
            user_agent_header=WA_USER_AGENT,
            compression=None,
            ping_interval=None,
            ping_timeout=None,
        ) as ws:
            self._ws = ws

            ephemeral_priv, _, ephemeral_pub = generate_x25519_keypair()
            noise = NoiseHandshakeState()
            noise.start(WA_HEADER)
            noise.authenticate(ephemeral_pub)

            hello = HandshakeMessage()
            hello.client_hello.ephemeral = ephemeral_pub
            await ws.send(self._build_frame(hello.SerializeToString(), include_header=True))

            resp = await asyncio.wait_for(ws.recv(), timeout=10)
            if isinstance(resp, str):
                resp = resp.encode("latin-1")

            frame_len = int.from_bytes(resp[:3], "big")
            server_hello = HandshakeMessage()
            server_hello.ParseFromString(resp[3 : 3 + frame_len])
            sh = server_hello.server_hello

            noise.authenticate(sh.ephemeral)
            server_ephemeral_pub = x25519.X25519PublicKey.from_public_bytes(sh.ephemeral)
            noise.mix_key(ephemeral_priv.exchange(server_ephemeral_pub))

            server_static_bytes = noise.decrypt(sh.static)
            server_static_pub = x25519.X25519PublicKey.from_public_bytes(server_static_bytes)
            noise.mix_key(ephemeral_priv.exchange(server_static_pub))
            noise.decrypt(sh.payload)

            enc_static = noise.encrypt(self.pairing.static_pub)
            noise.mix_key(self.pairing.static_priv.exchange(server_ephemeral_pub))
            enc_payload = noise.encrypt(self.pairing.build_client_payload())

            finish_msg = HandshakeMessage()
            finish_msg.client_finish.static = enc_static
            finish_msg.client_finish.payload = enc_payload
            await ws.send(self._build_frame(finish_msg.SerializeToString(), include_header=False))

            self._transport = noise.finish()

            buffer = bytearray()

            try:
                while True:
                    chunk = await asyncio.wait_for(ws.recv(), timeout=90)
                    if isinstance(chunk, str):
                        chunk = chunk.encode("latin-1")
                    buffer.extend(chunk)

                    while len(buffer) >= 3:
                        length = int.from_bytes(buffer[:3], "big")
                        if len(buffer) < 3 + length:
                            break
                        encrypted_frame = bytes(buffer[3 : 3 + length])
                        del buffer[: 3 + length]

                        decrypted = self._transport.decrypt(encrypted_frame)
                        node = decode_binary_node(self._unpack_transport(decrypted))

                        if self.debug:
                            print(f"[OK] {node}")
                        await self._emit("node", node)

                        if node.tag == "success":
                            if keepalive_task is None:
                                keepalive_task = asyncio.create_task(
                                    keepalive_loop(ws, self._transport, self.debug)
                                )
                            await self._emit("ready", self.pairing.jid)
                            continue

                        reconnect = await handle_node(node, self)
                        if reconnect:
                            should_reconnect = True
                            return True

            except asyncio.TimeoutError:
                print("[*] Timeout esperando actividad. Reconectando...")
                should_reconnect = True
            except websockets.ConnectionClosed as e:
                print(f"[-] Conexión cerrada: {e}. Reconectando...")
                should_reconnect = True
            finally:
                if keepalive_task is not None:
                    keepalive_task.cancel()
                for fut in self._pending_iqs.values():
                    if not fut.done():
                        fut.cancel()
                self._pending_iqs.clear()
                self._ws = None
                self._transport = None

        return should_reconnect

    async def start(self):
        while True:
            try:
                should_reconnect = await self._connect_once()
                if not should_reconnect:
                    break
            except (OSError, asyncio.TimeoutError, websockets.WebSocketException) as e:
                print(f"[!] Error de conexión ({type(e).__name__}: {e}). Reintento en 3 segundos...")
                await asyncio.sleep(3)
                continue
            await asyncio.sleep(1)

# Serialización y deserialización del protocolo binario de WhatsApp (WABinary)
from dataclasses import dataclass, field
from typing import Any
from binary.tokens import DOUBLE_BYTE_TOKENS, SINGLE_BYTE_TOKENS

TAGS = {
    "LIST_EMPTY": 0,
    "DICTIONARY_0": 236,
    "DICTIONARY_1": 237,
    "DICTIONARY_2": 238,
    "DICTIONARY_3": 239,
    "AD_JID": 247,
    "LIST_8": 248,
    "LIST_16": 249,
    "JID_PAIR": 250,
    "HEX_8": 251,
    "BINARY_8": 252,
    "BINARY_20": 253,
    "BINARY_32": 254,
    "NIBBLE_8": 255,
}

SINGLE_TOKEN_MAP = {tok: i for i, tok in enumerate(SINGLE_BYTE_TOKENS) if tok}
DOUBLE_TOKEN_MAP: dict[str, tuple[int, int]] = {}
for dict_idx, table in enumerate(DOUBLE_BYTE_TOKENS):
    for tok_idx, tok in enumerate(table):
        if tok and tok not in DOUBLE_TOKEN_MAP:
            DOUBLE_TOKEN_MAP[tok] = (dict_idx, tok_idx)

@dataclass
class BinaryNode:
    tag: str
    attrs: dict[str, str] = field(default_factory=dict)
    content: Any = None

    def get_child(self, tag: str) -> "BinaryNode | None":
        if isinstance(self.content, list):
            for child in self.content:
                if isinstance(child, BinaryNode) and child.tag == tag:
                    return child
        return None

    def get_children(self, tag: str) -> list["BinaryNode"]:
        if isinstance(self.content, list):
            return [c for c in self.content if isinstance(c, BinaryNode) and c.tag == tag]
        return []

    def __repr__(self) -> str:
        attrs_str = " ".join(f'{k}="{v}"' for k, v in self.attrs.items())
        header = f"<{self.tag} {attrs_str}>" if attrs_str else f"<{self.tag}>"
        if self.content is None:
            return header[:-1] + "/>"
        if isinstance(self.content, list):
            inner = ", ".join(repr(c) for c in self.content)
            return f"{header}[{inner}]</{self.tag}>"
        if isinstance(self.content, bytes):
            try:
                text = self.content.decode("utf-8")
                if len(text) > 40:
                    text = text[:40] + "..."
                return f"{header}{text!r}</{self.tag}>"
            except UnicodeDecodeError:
                return f"{header}<{len(self.content)} bytes></{self.tag}>"
        return f"{header}{self.content!r}</{self.tag}>"

class _Decoder:
    def __init__(self, data: bytes):
        self.data = data
        self.pos = 0

    def read_byte(self) -> int:
        b = self.data[self.pos]
        self.pos += 1
        return b

    def read_bytes(self, n: int) -> bytes:
        chunk = self.data[self.pos : self.pos + n]
        self.pos += n
        return chunk

    def read_int(self, n: int) -> int:
        return int.from_bytes(self.read_bytes(n), "big")

    def read_int20(self) -> int:
        b0, b1, b2 = self.read_byte(), self.read_byte(), self.read_byte()
        return ((b0 & 0x0F) << 16) | (b1 << 8) | b2

    def read_packed8(self, tag: int) -> str:
        start_byte = self.read_byte()
        length = start_byte & 127
        is_odd = bool(start_byte & 128)
        raw = self.read_bytes(length)

        chars = []
        for b in raw:
            for nib in ((b >> 4) & 0x0F, b & 0x0F):
                if tag == TAGS["NIBBLE_8"]:
                    if 0 <= nib <= 9:
                        chars.append(chr(48 + nib))
                    elif nib == 10:
                        chars.append("-")
                    elif nib == 11:
                        chars.append(".")
                    elif nib == 15:
                        chars.append("\x00")
                else:
                    if 0 <= nib <= 9:
                        chars.append(chr(48 + nib))
                    elif 10 <= nib <= 15:
                        chars.append(chr(65 + nib - 10))
        if is_odd and chars:
            chars.pop()
        return "".join(c for c in chars if c != "\x00")

    def read_list_size(self, tag: int) -> int:
        if tag == TAGS["LIST_EMPTY"]:
            return 0
        if tag == TAGS["LIST_8"]:
            return self.read_byte()
        if tag == TAGS["LIST_16"]:
            return self.read_int(2)
        raise ValueError(f"Tag de lista inválido: {tag}")

    def read_string_from_tag(self, tag: int) -> str:
        if 1 <= tag < len(SINGLE_BYTE_TOKENS):
            return SINGLE_BYTE_TOKENS[tag]
        if TAGS["DICTIONARY_0"] <= tag <= TAGS["DICTIONARY_3"]:
            dict_idx = tag - TAGS["DICTIONARY_0"]
            tok_idx = self.read_byte()
            return DOUBLE_BYTE_TOKENS[dict_idx][tok_idx]
        if tag == TAGS["LIST_EMPTY"]:
            return ""
        if tag == TAGS["BINARY_8"]:
            return self.read_bytes(self.read_byte()).decode("utf-8", errors="replace")
        if tag == TAGS["BINARY_20"]:
            return self.read_bytes(self.read_int20()).decode("utf-8", errors="replace")
        if tag == TAGS["BINARY_32"]:
            return self.read_bytes(self.read_int(4)).decode("utf-8", errors="replace")
        if tag == TAGS["JID_PAIR"]:
            user = self.read_string_from_tag(self.read_byte())
            server = self.read_string_from_tag(self.read_byte())
            return f"{user}@{server}" if user else server
        if tag == TAGS["AD_JID"]:
            domain_type = self.read_byte()
            device = self.read_byte()
            user = self.read_string_from_tag(self.read_byte())
            domain = "lid" if domain_type == 1 else "s.whatsapp.net"
            return f"{user}:{device}@{domain}" if device else f"{user}@{domain}"
        if tag in (TAGS["NIBBLE_8"], TAGS["HEX_8"]):
            return self.read_packed8(tag)
        raise ValueError(f"Tag de string desconocido: {tag}")

    def read_node(self) -> BinaryNode:
        list_size = self.read_list_size(self.read_byte())
        if list_size == 0:
            raise ValueError("Nodo vacío inválido")

        tag = self.read_string_from_tag(self.read_byte())
        attrs: dict[str, str] = {}
        num_attrs = (list_size - 1) >> 1
        for _ in range(num_attrs):
            key = self.read_string_from_tag(self.read_byte())
            val = self.read_string_from_tag(self.read_byte())
            attrs[key] = val

        content: Any = None
        if list_size % 2 == 0:
            content_tag = self.read_byte()
            if content_tag in (TAGS["LIST_EMPTY"], TAGS["LIST_8"], TAGS["LIST_16"]):
                child_count = self.read_list_size(content_tag)
                content = [self.read_node() for _ in range(child_count)]
            elif content_tag == TAGS["BINARY_8"]:
                content = self.read_bytes(self.read_byte())
            elif content_tag == TAGS["BINARY_20"]:
                content = self.read_bytes(self.read_int20())
            elif content_tag == TAGS["BINARY_32"]:
                content = self.read_bytes(self.read_int(4))
            else:
                content = self.read_string_from_tag(content_tag)

        return BinaryNode(tag=tag, attrs=attrs, content=content)

def decode_binary_node(data: bytes) -> BinaryNode:
    return _Decoder(data).read_node()

class _Encoder:
    def __init__(self):
        self.buf = bytearray()

    def write_byte(self, b: int):
        self.buf.append(b & 0xFF)

    def write_bytes(self, data: bytes):
        length = len(data)
        if length < 256:
            self.write_byte(TAGS["BINARY_8"])
            self.write_byte(length)
        elif length < (1 << 20):
            self.write_byte(TAGS["BINARY_20"])
            self.write_byte((length >> 16) & 0x0F)
            self.write_byte((length >> 8) & 0xFF)
            self.write_byte(length & 0xFF)
        else:
            self.write_byte(TAGS["BINARY_32"])
            self.buf.extend(length.to_bytes(4, "big"))
        self.buf.extend(data)

    def write_string(self, s: str):
        if s in SINGLE_TOKEN_MAP:
            self.write_byte(SINGLE_TOKEN_MAP[s])
            return
        if s in DOUBLE_TOKEN_MAP:
            dict_idx, tok_idx = DOUBLE_TOKEN_MAP[s]
            self.write_byte(TAGS["DICTIONARY_0"] + dict_idx)
            self.write_byte(tok_idx)
            return
        if "@" in s:
            user, server = s.split("@", 1)
            if ":" in user and server in ("s.whatsapp.net", "lid"):
                base_user, dev_str = user.split(":", 1)
                if dev_str.isdigit():
                    domain_type = 1 if server == "lid" else 0
                    self.write_byte(TAGS["AD_JID"])
                    self.write_byte(domain_type)
                    self.write_byte(int(dev_str))
                    self.write_string(base_user)
                    return

            self.write_byte(TAGS["JID_PAIR"])
            if user:
                self.write_string(user)
            else:
                self.write_byte(TAGS["LIST_EMPTY"])
            self.write_string(server)
            return
        self.write_bytes(s.encode("utf-8"))

    def write_list_start(self, length: int):
        if length == 0:
            self.write_byte(TAGS["LIST_EMPTY"])
        elif length < 256:
            self.write_byte(TAGS["LIST_8"])
            self.write_byte(length)
        else:
            self.write_byte(TAGS["LIST_16"])
            self.buf.extend(length.to_bytes(2, "big"))

    def write_node(self, node: BinaryNode):
        has_content = node.content is not None
        list_size = 1 + (2 * len(node.attrs)) + (1 if has_content else 0)
        self.write_list_start(list_size)
        self.write_string(node.tag)

        for k, v in node.attrs.items():
            self.write_string(k)
            self.write_string(str(v))

        if has_content:
            if isinstance(node.content, bytes):
                self.write_bytes(node.content)
            elif isinstance(node.content, str):
                self.write_string(node.content)
            elif isinstance(node.content, list):
                self.write_list_start(len(node.content))
                for child in node.content:
                    self.write_node(child)

def encode_binary_node(node: BinaryNode) -> bytes:
    enc = _Encoder()
    enc.write_node(node)
    return bytes(enc.buf)

# Handshake Noise XX: negociación de claves efímeras y estáticas con AES-GCM + SHA-256
import hashlib
import struct
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.ciphers.aead import AESGCM
from cryptography.hazmat.primitives.kdf.hkdf import HKDF

from .noise import NoiseTransport

NOISE_PATTERN = b"Noise_XX_25519_AESGCM_SHA256\x00\x00\x00\x00"

class NoiseHandshakeState:

    def __init__(self):
        self.h = NOISE_PATTERN
        self.ck = NOISE_PATTERN
        self.key = NOISE_PATTERN
        self.counter = 0

    def mix_hash(self, data: bytes):
        self.h = hashlib.sha256(self.h + data).digest()

    def start(self, header: bytes):
        self.mix_hash(header)

    def authenticate(self, data: bytes):
        self.mix_hash(data)

    def _generate_iv(self) -> bytes:
        iv = b"\x00" * 8 + struct.pack(">I", self.counter)
        self.counter += 1
        return iv

    def mix_key(self, data: bytes):
        hkdf = HKDF(
            algorithm=hashes.SHA256(),
            length=64,
            salt=self.ck,
            info=b"",
        )
        derived = hkdf.derive(data)
        self.ck = derived[:32]
        self.key = derived[32:]
        self.counter = 0

    def encrypt(self, plaintext: bytes) -> bytes:
        iv = self._generate_iv()
        ciphertext = AESGCM(self.key).encrypt(iv, plaintext, self.h)
        self.mix_hash(ciphertext)
        return ciphertext

    def decrypt(self, ciphertext: bytes) -> bytes:
        iv = self._generate_iv()
        plaintext = AESGCM(self.key).decrypt(iv, ciphertext, self.h)
        self.mix_hash(ciphertext)
        return plaintext

    def finish(self) -> NoiseTransport:
        hkdf = HKDF(
            algorithm=hashes.SHA256(),
            length=64,
            salt=self.ck,
            info=b"",
        )
        derived = hkdf.derive(b"")
        return NoiseTransport(write_key=derived[:32], read_key=derived[32:])

# Generación de claves X25519 y firma XEdDSA (Signal/WhatsApp Curve25519)
import hashlib
import os
from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.primitives.asymmetric import x25519

_P = (1 << 255) - 19
_L = (1 << 252) + 27742317777372353535851937790883648493
_D = (-121665 * pow(121666, _P - 2, _P)) % _P

_BX = 15112221349535400772501151409588531511454012693041857206046113283949847762202
_BY = 46316835694926478169428394003475163141307993866256225615783033603165251855960
_BASE = (_BX, _BY, 1, (_BX * _BY) % _P)
_IDENTITY = (0, 1, 1, 0)

def _point_add(
    p1: tuple[int, int, int, int], p2: tuple[int, int, int, int]
) -> tuple[int, int, int, int]:
    a = ((p1[1] - p1[0]) * (p2[1] - p2[0])) % _P
    b = ((p1[1] + p1[0]) * (p2[1] + p2[0])) % _P
    c = (2 * p1[3] * p2[3] * _D) % _P
    d = (2 * p1[2] * p2[2]) % _P
    e = (b - a) % _P
    f = (d - c) % _P
    g = (d + c) % _P
    h = (b + a) % _P
    return ((e * f) % _P, (g * h) % _P, (f * g) % _P, (e * h) % _P)

def _scalar_mult(s: int, point: tuple[int, int, int, int]) -> tuple[int, int, int, int]:
    res = _IDENTITY
    addend = point
    while s > 0:
        if s & 1:
            res = _point_add(res, addend)
        addend = _point_add(addend, addend)
        s >>= 1
    return res

def _encode_point(point: tuple[int, int, int, int]) -> bytes:
    zinv = pow(point[2], _P - 2, _P)
    x = (point[0] * zinv) % _P
    y = (point[1] * zinv) % _P
    out = bytearray(y.to_bytes(32, "little"))
    out[31] |= (x & 1) << 7
    return bytes(out)

def generate_x25519_keypair() -> tuple[x25519.X25519PrivateKey, bytes, bytes]:
    priv = x25519.X25519PrivateKey.generate()
    priv_bytes = priv.private_bytes(
        encoding=serialization.Encoding.Raw,
        format=serialization.PrivateFormat.Raw,
        encryption_algorithm=serialization.NoEncryption(),
    )
    pub_bytes = priv.public_key().public_bytes(
        encoding=serialization.Encoding.Raw,
        format=serialization.PublicFormat.Raw,
    )
    return priv, priv_bytes, pub_bytes

def xeddsa_sign(private_key_bytes: bytes, message: bytes) -> bytes:
    k = bytearray(private_key_bytes)
    k[0] &= 248
    k[31] &= 127
    k[31] |= 64
    a = int.from_bytes(k, "little")

    e_point = _scalar_mult(a, _BASE)
    zinv = pow(e_point[2], _P - 2, _P)
    ex = (e_point[0] * zinv) % _P
    ey = (e_point[1] * zinv) % _P

    if ex & 1:
        a = (-a) % _L
    else:
        a = a % _L

    a_pub_bytes = ey.to_bytes(32, "little")
    a_priv_bytes = a.to_bytes(32, "little")

    nonce_z = os.urandom(64)
    prefix = b"\xfe" + b"\xff" * 31
    r = int.from_bytes(
        hashlib.sha512(prefix + a_priv_bytes + message + nonce_z).digest(), "little"
    ) % _L

    r_point = _scalar_mult(r, _BASE)
    r_bytes = _encode_point(r_point)

    h = int.from_bytes(
        hashlib.sha512(r_bytes + a_pub_bytes + message).digest(), "little"
    ) % _L
    s = (r + h * a) % _L

    return r_bytes + s.to_bytes(32, "little")

class NoiseTransport:

    def __init__(self, write_key: bytes, read_key: bytes):
        self.write_cipher = AESGCM(write_key)
        self.read_cipher = AESGCM(read_key)
        self.write_counter = 0
        self.read_counter = 0

    def _make_iv(self, counter: int) -> bytes:
        return b"\x00" * 8 + struct.pack(">I", counter)

    def encrypt(self, plaintext: bytes) -> bytes:
        iv = self._make_iv(self.write_counter)
        self.write_counter += 1
        return self.write_cipher.encrypt(iv, plaintext, b"")

    def decrypt(self, ciphertext: bytes) -> bytes:
        iv = self._make_iv(self.read_counter)
        self.read_counter += 1
        return self.read_cipher.decrypt(iv, ciphertext, b"")

def _b64e(data: bytes) -> str:
    return base64.b64encode(data).decode("ascii")


def _b64d(text: str) -> bytes:
    return base64.b64decode(text.encode("ascii"))


def _strip_key_prefix(key_bytes: bytes) -> bytes:
    return key_bytes[1:] if len(key_bytes) == 33 and key_bytes[0] == 0x05 else key_bytes


def _add_key_prefix(key_bytes: bytes) -> bytes:
    return key_bytes if len(key_bytes) == 33 else (b"\x05" + key_bytes)


def _dh(priv_bytes: bytes, pub_bytes: bytes) -> bytes:
    priv = x25519.X25519PrivateKey.from_private_bytes(priv_bytes)
    pub = x25519.X25519PublicKey.from_public_bytes(_strip_key_prefix(pub_bytes))
    return priv.exchange(pub)


def _hkdf_sha256(ikm: bytes, salt: bytes, info: bytes, length: int) -> bytes:
    return HKDF(
        algorithm=hashes.SHA256(),
        length=length,
        salt=salt,
        info=info,
    ).derive(ikm)


def _hmac_sha256(key: bytes, data: bytes) -> bytes:
    return hmac.new(key, data, hashlib.sha256).digest()


def _aes_cbc_decrypt(key: bytes, iv: bytes, ciphertext: bytes) -> bytes:
    cipher = Cipher(algorithms.AES(key), modes.CBC(iv))
    decryptor = cipher.decryptor()
    padded = decryptor.update(ciphertext) + decryptor.finalize()
    unpadder = sym_padding.PKCS7(128).unpadder()
    return unpadder.update(padded) + unpadder.finalize()


def _aes_cbc_encrypt(key: bytes, iv: bytes, plaintext: bytes) -> bytes:
    padder = sym_padding.PKCS7(128).padder()
    padded = padder.update(plaintext) + padder.finalize()
    cipher = Cipher(algorithms.AES(key), modes.CBC(iv))
    encryptor = cipher.encryptor()
    return encryptor.update(padded) + encryptor.finalize()


def _unpad_wa_plaintext(data: bytes) -> bytes:
    if not data:
        return data
    pad_len = data[-1]
    if 0 < pad_len <= len(data):
        return data[:-pad_len]
    return data


def _pad_wa_plaintext(data: bytes) -> bytes:
    pad_len = random.randint(1, 16)
    return data + bytes([pad_len]) * pad_len


# ─── Sender Key Chain helpers (grupos) ──────────────────────────────────────

def _derive_sender_key_step(chain_key: bytes) -> tuple[bytes, bytes]:
    """
    Avanza la cadena de sender key un paso.
    Retorna (message_key_seed, next_chain_key).
    """
    msg_key = _hmac_sha256(chain_key, b"\x01")
    next_chain = _hmac_sha256(chain_key, b"\x02")
    return msg_key, next_chain


def _sender_msg_key_to_cipher_keys(msg_key: bytes) -> tuple[bytes, bytes]:
    """
    Expande el message_key_seed de grupo a (cipher_key 32B, iv 16B).
    En libsignal / Baileys / Whatsmeow (WhisperGroup):
      bytes  0..16 = IV (16 bytes)
      bytes 16..48 = Cipher Key (32 bytes)
    """
    derived = _hkdf_sha256(msg_key, salt=b"\x00" * 32, info=b"WhisperGroup", length=48)
    iv = derived[:16]
    cipher_key = derived[16:48]
    return cipher_key, iv


def _encode_varint(n: int) -> bytes:
    out = bytearray()
    while True:
        b = n & 0x7F
        n >>= 7
        if n:
            out.append(b | 0x80)
        else:
            out.append(b)
            break
    return bytes(out)


class SignalSessionManager:

    def __init__(
        self,
        ident_priv_bytes: bytes,
        ident_pub_bytes: bytes,
        skey_priv_bytes: bytes,
        prekeys,
        sessions_path: Path = SESSIONS_PATH,

        self.ident_pub_bytes = ident_pub_bytes
        self.skey_priv_bytes = skey_priv_bytes
        self.prekeys = prekeys
        self.sessions_path = sessions_path
        self.sender_keys_path = sender_keys_path

        self.sessions: dict[str, dict] = {}
        if self.sessions_path.exists():
            self.sessions = json.loads(self.sessions_path.read_text(encoding="utf-8"))

        # sender_keys[group_jid][sender_jid] = {id, chain_key, iteration, signing_key_pub}
        self.sender_keys: dict[str, dict[str, dict]] = {}
        if self.sender_keys_path.exists():
            self.sender_keys = json.loads(self.sender_keys_path.read_text(encoding="utf-8"))

    def _save_sessions(self):
        self.sessions_path.parent.mkdir(parents=True, exist_ok=True)
        self.sessions_path.write_text(json.dumps(self.sessions, indent=2), encoding="utf-8")

    def _save_sender_keys(self):
        self.sender_keys_path.parent.mkdir(parents=True, exist_ok=True)
        self.sender_keys_path.write_text(json.dumps(self.sender_keys, indent=2), encoding="utf-8")

    # ─── Sender Key Distribution ─────────────────────────────────────────────

    def process_sender_key_distribution(
        self,
        group_jid: str,
        sender_jid: str,
        skdm_bytes: bytes,
    ) -> None:
        """
        Procesa un SenderKeyDistributionMessage y guarda la sender key
        para poder descifrar mensajes de grupo de ese sender al instante.
        """
        try:
            skdm = SenderKeyDistributionMessage()
            skdm.ParseFromString(skdm_bytes)
        except Exception as e:
            print(f"[!] No se pudo parsear SenderKeyDistributionMessage de {sender_jid}: {e}")
            return

        if not skdm.chain_key:
            print(f"[!] SenderKeyDistributionMessage sin chain_key de {sender_jid}")
            return

        group_map = self.sender_keys.setdefault(group_jid, {})
        group_map[sender_jid] = {
            "id": skdm.id,
            "chain_key": _b64e(skdm.chain_key),
            "iteration": skdm.iteration,
            "signing_key_pub": _b64e(skdm.signing_key),
        }
        self._save_sender_keys()
        print(
            f"[*] Sender key de {sender_jid} en grupo {group_jid} registrada "
            f"(id={skdm.id}, iter={skdm.iteration})"
        )

    # ─── Descifrado de Sender Key Message (grupos) ───────────────────────────

    def decrypt_skmsg(self, group_jid: str, sender_jid: str, raw_skmsg: bytes) -> bytes:
        """
        Descifra un SenderKeyMessage (enc type="skmsg") de un grupo.
        """
        base_group = group_jid.split("@")[0].split(":")[0] + "@" + group_jid.split("@")[1]
        base_sender = sender_jid.split("@")[0].split(":")[0] + "@" + (
            sender_jid.split("@")[1] if "@" in sender_jid else "s.whatsapp.net"
        )

        group_map = self.sender_keys.get(group_jid) or self.sender_keys.get(base_group, {})
        sk_data = group_map.get(sender_jid) or group_map.get(base_sender)

        if not sk_data:
            raise ValueError(
                f"No hay sender key guardada para {sender_jid} en {group_jid}. "
                "Esperando SenderKeyDistributionMessage."
            )

        # Formato libsignal: 1 byte versión (0x33) + proto + 64 bytes firma XEdDSA
        if len(raw_skmsg) < 66 or raw_skmsg[0] != 0x33:
            raise ValueError(f"SenderKeyMessage inválido ({len(raw_skmsg)} bytes)")

        skm = SenderKeyMessage()
        try:
            skm.ParseFromString(raw_skmsg[1:-64])
        except Exception as e:
            raise ValueError(f"Error parseando SenderKeyMessage: {e}") from e

        iteration = skm.iteration
        ciphertext = skm.ciphertext

        chain_key = _b64d(sk_data["chain_key"])
        saved_iter = int(sk_data["iteration"])

        if iteration < saved_iter:
            raise ValueError(
                f"Iteración {iteration} ya fue procesada (guardada={saved_iter}) para {sender_jid}"
            )

        for _ in range(iteration - saved_iter):
            _, chain_key = _derive_sender_key_step(chain_key)

        msg_key, next_chain = _derive_sender_key_step(chain_key)
        cipher_key, iv = _sender_msg_key_to_cipher_keys(msg_key)

        plaintext = _aes_cbc_decrypt(cipher_key, iv, ciphertext)
        plaintext = _unpad_wa_plaintext(plaintext)

        sk_data["chain_key"] = _b64e(next_chain)
        sk_data["iteration"] = iteration + 1
        for k in (group_jid, base_group):
            if k in self.sender_keys:
                for s in (sender_jid, base_sender):
                    if s in self.sender_keys[k]:
                        self.sender_keys[k][s] = sk_data
        self._save_sender_keys()

        return plaintext

    # ─── Cifrado de Sender Key Message (grupos) ──────────────────────────────

    def get_or_create_own_sender_key(self, group_jid: str) -> tuple[dict, bytes]:
        """
        Obtiene o crea nuestra propia SenderKey para un grupo y devuelve
        (sk_data, axolotl_skdm_bytes) listos para distribuir a los participantes.
        """
        group_map = self.sender_keys.setdefault(group_jid, {})
        sk = group_map.get("__self__")
        if not sk:
            _, sig_priv, sig_pub = generate_x25519_keypair()
            sk = {
                "id": random.randint(1, 2147483647),
                "iteration": 0,
                "chain_key": _b64e(os.urandom(32)),
                "signing_key_priv": _b64e(sig_priv),
                "signing_key_pub": _b64e(sig_pub),
            }
            group_map["__self__"] = sk
            self._save_sender_keys()

        proto = (
            _encode_varint((1 << 3) | 0) + _encode_varint(sk["id"])
            + _encode_varint((2 << 3) | 0) + _encode_varint(sk["iteration"])
            + _encode_varint((3 << 3) | 2) + _encode_varint(32) + _b64d(sk["chain_key"])
            + _encode_varint((4 << 3) | 2)
            + _encode_varint(33)
            + _add_key_prefix(_b64d(sk["signing_key_pub"]))
        )
        return sk, b"\x33" + proto

    def encrypt_skmsg(self, group_jid: str, plaintext_proto: bytes) -> tuple[bytes, bytes]:
        """
        Cifra un WAMessage para un grupo usando nuestra SenderKey.
        Retorna (skmsg_bytes, axolotl_skdm_bytes).
        """
        sk, skdm_bytes = self.get_or_create_own_sender_key(group_jid)
        chain_key = _b64d(sk["chain_key"])
        iteration = int(sk["iteration"])

        msg_key, next_chain = _derive_sender_key_step(chain_key)
        cipher_key, iv = _sender_msg_key_to_cipher_keys(msg_key)

        padded = _pad_wa_plaintext(plaintext_proto)
        ciphertext = _aes_cbc_encrypt(cipher_key, iv, padded)

        proto = (
            _encode_varint((1 << 3) | 0) + _encode_varint(sk["id"])
            + _encode_varint((2 << 3) | 0) + _encode_varint(iteration)
            + _encode_varint((3 << 3) | 2) + _encode_varint(len(ciphertext)) + ciphertext
        )
        payload = b"\x33" + proto
        sig = xeddsa_sign(_b64d(sk["signing_key_priv"]), payload)

        sk["chain_key"] = _b64e(next_chain)
        sk["iteration"] = iteration + 1
        self.sender_keys[group_jid]["__self__"] = sk
        self._save_sender_keys()

        return payload + sig, skdm_bytes

    # ─── Signal (pkmsg / msg) ────────────────────────────────────────────────

    def _decrypt_signal_message(self, jid: str, raw_msg: bytes, session: dict) -> bytes:
        if len(raw_msg) < 10 or raw_msg[0] != 0x33:
            raise ValueError(f"Versión de SignalMessage inválida: {raw_msg[:1].hex()}")

        body = raw_msg[1:-8]
        sig_msg = SignalMessage()
        sig_msg.ParseFromString(body)

        their_ratchet = _strip_key_prefix(sig_msg.ratchet_key)
        counter = sig_msg.counter

        root_key = _b64d(session["root_key"])
        remote_ratchet = _b64d(session["remote_ratchet"]) if session.get("remote_ratchet") else None
        recv_chain_key = _b64d(session["recv_chain_key"]) if session.get("recv_chain_key") else None
        recv_counter = int(session.get("recv_counter", 0))

        if remote_ratchet != their_ratchet or recv_chain_key is None:
            local_priv = (
                _b64d(session["local_ratchet_priv"])
                if session.get("local_ratchet_priv")
                else self.skey_priv_bytes
            )
            dh_out = _dh(local_priv, their_ratchet)
            derived = _hkdf_sha256(dh_out, salt=root_key, info=b"WhisperRatchet", length=64)
            root_key = derived[:32]
            recv_chain_key = derived[32:64]
            remote_ratchet = their_ratchet
            recv_counter = 0
            session["send_chain_key"] = None
            session["previous_counter"] = int(session.get("send_counter", 0))
            session["send_counter"] = 0

        if counter < recv_counter:
            raise ValueError(f"Contador atrasado ({counter} < {recv_counter}) para {jid}")

        chain_key = recv_chain_key
        for _ in range(counter - recv_counter):
            chain_key = _hmac_sha256(chain_key, b"\x02")

        msg_key_seed = _hmac_sha256(chain_key, b"\x01")
        next_chain_key = _hmac_sha256(chain_key, b"\x02")

        keys = _hkdf_sha256(msg_key_seed, salt=b"\x00" * 32, info=b"WhisperMessageKeys", length=80)
        cipher_key = keys[0:32]
        iv = keys[64:80]

        plaintext_padded = _aes_cbc_decrypt(cipher_key, iv, sig_msg.ciphertext)
        plaintext = _unpad_wa_plaintext(plaintext_padded)

        session["root_key"] = _b64e(root_key)
        session["remote_ratchet"] = _b64e(remote_ratchet)
        session["recv_chain_key"] = _b64e(next_chain_key)
        session["recv_counter"] = counter + 1
        self.sessions[jid] = session
        self._save_sessions()

        return plaintext

    def decrypt_pkmsg(self, jid: str, raw_pkmsg: bytes) -> bytes:
        if len(raw_pkmsg) < 2 or raw_pkmsg[0] != 0x33:
            raise ValueError(f"Versión de PreKeySignalMessage inválida: {raw_pkmsg[:1].hex()}")

        pkmsg = PreKeySignalMessage()
        pkmsg.ParseFromString(raw_pkmsg[1:])

        dh1 = _dh(self.skey_priv_bytes, pkmsg.identity_key)
        dh2 = _dh(self.ident_priv_bytes, pkmsg.base_key)
        dh3 = _dh(self.skey_priv_bytes, pkmsg.base_key)
        master_secret = (b"\xff" * 32) + dh1 + dh2 + dh3

        if pkmsg.HasField("pre_key_id"):
            opk_priv = self.prekeys.get_private_key(pkmsg.pre_key_id)
            if opk_priv:
                master_secret += _dh(opk_priv, pkmsg.base_key)

        derived = _hkdf_sha256(master_secret, salt=b"\x00" * 32, info=b"WhisperText", length=64)
        session = {
            "remote_identity": _b64e(_strip_key_prefix(pkmsg.identity_key)),
            "root_key": _b64e(derived[:32]),
            "remote_ratchet": None,
            "local_ratchet_priv": None,
            "local_ratchet_pub": None,
            "recv_chain_key": None,
            "recv_counter": 0,
            "send_chain_key": None,
            "send_counter": 0,
            "previous_counter": 0,
        }

        return self._decrypt_signal_message(jid, pkmsg.message, session)

    def decrypt_msg(self, jid: str, raw_msg: bytes) -> bytes:
        session = self.sessions.get(jid)
        if not session:
            raise ValueError(f"No hay sesión Signal guardada para {jid}")
        return self._decrypt_signal_message(jid, raw_msg, session)

    def encrypt_msg(self, jid: str, plaintext_proto: bytes) -> bytes:
        session = self.sessions.get(jid)
        if not session or not session.get("remote_identity") or not session.get("remote_ratchet"):
            raise ValueError(
                f"No hay sesión Signal completa para {jid}. "
                "Primero debe llegar un mensaje de ese contacto."
            )

        root_key = _b64d(session["root_key"])
        remote_ratchet = _b64d(session["remote_ratchet"])
        remote_identity = _b64d(session["remote_identity"])

        if not session.get("send_chain_key"):
            _, local_priv, local_pub = generate_x25519_keypair()
            dh_out = _dh(local_priv, remote_ratchet)
            derived = _hkdf_sha256(dh_out, salt=root_key, info=b"WhisperRatchet", length=64)
            root_key = derived[:32]
            send_chain_key = derived[32:64]
            send_counter = 0
            session["root_key"] = _b64e(root_key)
            session["local_ratchet_priv"] = _b64e(local_priv)
            session["local_ratchet_pub"] = _b64e(local_pub)
        else:
            local_pub = _b64d(session["local_ratchet_pub"])
            send_chain_key = _b64d(session["send_chain_key"])
            send_counter = int(session.get("send_counter", 0))

        msg_key_seed = _hmac_sha256(send_chain_key, b"\x01")
        next_send_chain_key = _hmac_sha256(send_chain_key, b"\x02")

        keys = _hkdf_sha256(msg_key_seed, salt=b"\x00" * 32, info=b"WhisperMessageKeys", length=80)
        cipher_key = keys[0:32]
        mac_key = keys[32:64]
        iv = keys[64:80]

        padded_wa = _pad_wa_plaintext(plaintext_proto)
        ciphertext = _aes_cbc_encrypt(cipher_key, iv, padded_wa)

        sig_msg = SignalMessage()
        sig_msg.ratchet_key = _add_key_prefix(local_pub)
        sig_msg.counter = send_counter
        sig_msg.previous_counter = int(session.get("previous_counter", 0))
        sig_msg.ciphertext = ciphertext
        body = sig_msg.SerializeToString()

        mac_input = (
            _add_key_prefix(self.ident_pub_bytes)
            + _add_key_prefix(remote_identity)
            + b"\x33"
            + body
        )
        mac = _hmac_sha256(mac_key, mac_input)[:8]

        session["send_chain_key"] = _b64e(next_send_chain_key)
        session["send_counter"] = send_counter + 1
        self.sessions[jid] = session
        self._save_sessions()

        return b"\x33" + body + mac

@client.on("message")
async def on_message(umsg: IncomingMessage):
    umsg_name = umsg.push_name or umsg.sender
    chat_jid = umsg.chat_jid          # grupo (@g.us) o JID del remitente
    tipo = "[GRUPO]" if umsg.is_group else "[CHAT]"

    print(f"{tipo} {umsg_name} ({umsg.sender}): {umsg.text}")

    if not umsg.text:
        return

    cmd = umsg.text.strip().lower()

    # ── !msg ─────────────────────────────────────────────────────────────────
    if cmd == "!msg":
        await client.send_msg(chat_jid, "¡Hola Mundo desde Snakechat! 🐍")

    # ── !reply ───────────────────────────────────────────────────────────────
    elif cmd == "!reply":
        await client.send_msg(chat_jid, "¡Te estoy respondiendo con quote! 💬", reply=umsg)

    # ── !forward ─────────────────────────────────────────────────────────────
    elif cmd == "!forward":
        await client.send_msg(chat_jid, "Mensaje con etiqueta de Reenviado ↪️", forward=True)

    # ── !mention ─────────────────────────────────────────────────────────────
    elif cmd == "!mention":
        await client.send_msg(chat_jid, f"¡Hola {tag_user(umsg)} con mención! 💚", tag=[umsg])

    # ── !mgroup ──────────────────────────────────────────────────────────────
    elif cmd == "!mgroup":
        await client.send_msg(chat_jid, f"Mención de grupo {tag_group(GROUP_JID)} 💙", g_tag=[(GROUP_JID, "Grupo SnakeChat")])

    # ── !edit ────────────────────────────────────────────────────────────────
    elif cmd == "!edit":
        original = await client.send_msg(chat_jid, "Mensaje original... (3 Segundos)")
        await asyncio.sleep(3)
        await client.edit_msg(chat_jid, original, "¡Mensaje editado con éxito!")

    # ── !delete ──────────────────────────────────────────────────────────────
    elif cmd == "!delete":
        if umsg.quoted: await client.delete_msg(chat_jid, umsg.quoted)

    # ── !read ────────────────────────────────────────────────────────────────
    elif cmd == "!read":
        await client.read_msg(umsg)

    # ── !typing ──────────────────────────────────────────────────────────────
    elif cmd == "!typing":
        await client.send_typing(chat_jid)
        await asyncio.sleep(3)
        await client.stop_typing(chat_jid)

    # ── !recording ───────────────────────────────────────────────────────────
    elif cmd == "!recording":
        await client.send_recording(chat_jid)
        await asyncio.sleep(3)
        await client.stop_recording(chat_jid)

    # ── !online / !offline ───────────────────────────────────────────────────
    elif cmd == "!online":
        await client.set_online()
    elif cmd == "!offline":
        await client.set_offline()

    # ── !react ───────────────────────────────────────────────────────────────
    elif cmd == "!react":
        await client.react_msg(chat_jid, "👍", target=umsg)

    # ── !sticker ─────────────────────────────────────────────────────────────
    elif cmd == "!sticker":
        await client.send_sticker(chat_jid, STICKER_PATH)

    # ── !sreact ──────────────────────────────────────────────────────────────
    elif cmd == "!sreact":
        await client.sticker_react(chat_jid, umsg, STICKER_PATH)

    # ── !info ────────────────────────────────────────────────────────────────
    elif cmd == "!info":
        info_text = (
            "ℹ️ *Información del usuario y mensaje*\n\n"
            f"• *Nombre:* {umsg.push_name or 'No disponible'}\n"
            f"• *Sender:* {umsg.sender}\n"
            f"• *Sender base:* {umsg.sender_base}\n"
            f"• *Chat JID:* {umsg.chat_jid}\n"
            f"• *Es grupo:* {'Sí' if umsg.is_group else 'No'}\n"
            f"• *LID:* {umsg.lid or 'No disponible'}\n"
            f"• *JID (PN):* {umsg.jid or 'No disponible'}\n"
            f"• *Número:* {umsg.number or 'No disponible'}\n"
            f"• *Dispositivo:* {umsg.device or 'Principal'}\n"
            f"• *Modo:* {umsg.addressing_mode or 'No especificado'}\n"
            f"• *ID Mensaje:* {umsg.id}"
        )
        await client.send_msg(chat_jid, info_text)

    # ── !help ────────────────────────────────────────────────────────────────
    elif cmd == "!help":
        help_text = (
            "🐍 *SnakeChat – Comandos disponibles*\n\n"
            "• !msg       ➜ mensaje de prueba\n"
            "• !reply     ➜ reply citando tu mensaje\n"
            "• !forward   ➜ mensaje con etiqueta Reenviado\n"
            "• !mention   ➜ mencionarte con @\n"
            "• !mgroup    ➜ mencionar grupo por ID\n"
            "• !edit      ➜ enviar mensaje y editarlo a los 3 s\n"
            "• !delete    ➜ borrar el mensaje citado\n"

import os


def _varint(n: int) -> bytes:
    if n < 0:
        n &= (1 << 64) - 1
    out = bytearray()
    while True:
        b = n & 0x7F
        n >>= 7
        if n:
            out.append(b | 0x80)
        else:
            out.append(b)
            break
    return bytes(out)


def _field_varint(field_num: int, val: int) -> bytes:
    return _varint((field_num << 3) | 0) + _varint(val)


def _field_bytes(field_num: int, val: bytes) -> bytes:
    return _varint((field_num << 3) | 2) + _varint(len(val)) + val


def _field_str(field_num: int, val: str) -> bytes:
    return _field_bytes(field_num, val.encode("utf-8"))


def _encode_message_key(
    remote_jid: str,
    msg_id: str,
    from_me: bool = False,
    participant: str | None = None,
) -> bytes:
    out = bytearray()
    if remote_jid:
        out.extend(_field_str(1, remote_jid))
    out.extend(_field_varint(2, 1 if from_me else 0))
    if msg_id:
        out.extend(_field_str(3, msg_id))
    if participant:
        out.extend(_field_str(4, participant))
    return bytes(out)

import struct


def _read_varint(data: bytes, pos: int) -> tuple[int, int]:
    result = 0
    shift = 0
    while True:
        if pos >= len(data):
            raise ValueError("Buffer terminó leyendo varint")
        b = data[pos]
        pos += 1
        result |= (b & 0x7F) << shift
        shift += 7
        if not (b & 0x80):
            break
    return result, pos


def _parse_proto_fields(data: bytes) -> dict:
    fields: dict = {}
    pos = 0
    while pos < len(data):
        tag_val, pos = _read_varint(data, pos)
        field_num = tag_val >> 3
        wire_type = tag_val & 0x07
        if wire_type == 0:
            val, pos = _read_varint(data, pos)
            fields[field_num] = val
        elif wire_type == 2:
            length, pos = _read_varint(data, pos)
            fields[field_num] = data[pos : pos + length]
            pos += length
        elif wire_type == 5:
            fields[field_num] = struct.unpack_from("<I", data, pos)[0]
            pos += 4
        elif wire_type == 1:
            fields[field_num] = struct.unpack_from("<Q", data, pos)[0]
            pos += 8
        else:
            raise ValueError(f"Wire type {wire_type} no soportado en pos {pos}")
    return fields


class SenderKeyMessage:
    def __init__(self):
        self.id: int = 0
        self.iteration: int = 0
        self.ciphertext: bytes = b""

    def ParseFromString(self, data: bytes) -> None:
        fields = _parse_proto_fields(data)
        self.id = int(fields.get(1, 0))
        self.iteration = int(fields.get(2, 0))
        self.ciphertext = bytes(fields.get(3, b""))

    value = int.from_bytes(data, "big")

    return kdf.derive(pairing_code.encode("utf-8"))

    def build_client_payload(self) -> bytes:
        cp = ClientPayload()
        self._fill_base_client_payload(cp)

        if self.is_registered and self.jid:
            user_part = self.jid.split("@")[0]
            if ":" in user_part:
                phone, dev_id = user_part.split(":", 1)
                cp.username = int(phone)
                cp.device = int(dev_id)
            else:
                cp.username = int(user_part)
            cp.passive = False
            cp.pull = True
            return cp.SerializeToString()

        dev_props = DeviceProps()
        dev_props.os = "Windows"
        dev_props.version.primary = 10
        dev_props.version.secondary = 0
        dev_props.version.tertiary = 0
        dev_props.platform_type = DeviceProps.CHROME
        dev_props.require_full_sync = False
        dev_props_bytes = dev_props.SerializeToString()

        skey_sig = xeddsa_sign(self.ident_priv_bytes, b"\x05" + self.skey_pub_bytes)
        build_hash = hashlib.md5(
            f"{self.wa_version[0]}.{self.wa_version[1]}.{self.wa_version[2]}".encode()
        ).digest()

        cp.passive = False
        cp.device_pairing_data.e_regid = struct.pack(">I", self.reg_id)
        cp.device_pairing_data.e_keytype = b"\x05"
        cp.device_pairing_data.e_ident = self.ident_pub_bytes
        cp.device_pairing_data.e_skey_id = struct.pack(">I", 1)[1:]
        cp.device_pairing_data.e_skey_val = self.skey_pub_bytes
        cp.device_pairing_data.e_skey_sig = skey_sig
        cp.device_pairing_data.build_hash = build_hash
        cp.device_pairing_data.device_props = dev_props_bytes

        return cp.SerializeToString()

    def build_pairing_code_iq(self, phone_number: str, custom_code: str | None = None) -> tuple[str, BinaryNode]:
        """
        Genera el código de 8 caracteres y construye el IQ stage='companion_hello'.
        Retorna (pairing_code, iq_node).
        """
        digits = "".join(ch for ch in phone_number if ch.isdigit())
        if not digits:
            raise ValueError("phone_number debe contener dígitos (ej: '50557418454').")

        code = (custom_code or _bytes_to_crockford(os.urandom(5))).strip().upper()
        if len(code) != 8:
            raise ValueError("El pairing_code personalizado debe tener exactamente 8 caracteres.")

        eph_priv, _, eph_pub = generate_x25519_keypair()
        self.pairing_code = code
        self.pairing_phone = digits
        self._pairing_ephemeral_priv = eph_priv
        self._pairing_ephemeral_pub = eph_pub

        salt = os.urandom(32)
        iv = os.urandom(16)
        enc_key = _derive_pairing_code_key(code, salt)
        encryptor = Cipher(algorithms.AES(enc_key), modes.CTR(iv)).encryptor()
        encrypted_pub = encryptor.update(eph_pub) + encryptor.finalize()
        wrapped_ephemeral_pub = salt + iv + encrypted_pub

        iq_node = BinaryNode(
            tag="iq",
            attrs={
                "to": "s.whatsapp.net",
                "type": "set",
                "id": "paircode-" + os.urandom(6).hex(),
                "xmlns": "md",
            },
            content=[
                BinaryNode(
                    tag="link_code_companion_reg",
                    attrs={
                        "jid": f"{digits}@s.whatsapp.net",
                        "stage": "companion_hello",
                        "should_show_push_notification": "true",
                    },
                    content=[
                        BinaryNode(
                            tag="link_code_pairing_wrapped_companion_ephemeral_pub",
                            content=wrapped_ephemeral_pub,
                        ),
                        BinaryNode(
                            tag="companion_server_auth_key_pub",
                            content=self.static_pub,
                        ),
                        BinaryNode(tag="companion_platform_id", content="1"),
                        BinaryNode(tag="companion_platform_display", content="Chrome (Windows)"),
                        BinaryNode(tag="link_code_pairing_nonce", content="0"),
                    ],
                )
            ],
        )
        return code, iq_node

    def handle_link_code_primary_hello(self, node: BinaryNode) -> BinaryNode | None:
        """
        Procesa <notification type='link_code_companion_reg'> con stage='primary_hello'
        y retorna el IQ stage='companion_finish'.
        """
        reg_node = node.get_child("link_code_companion_reg")
        if not reg_node or reg_node.attrs.get("stage") != "primary_hello":
            return None

        if not self.pairing_code or not self._pairing_ephemeral_priv or not self.pairing_phone:
            print("[!] Llegó primary_hello pero no hay un Pairing Code activo.")
            return None

        wrapped_node = reg_node.get_child("link_code_pairing_wrapped_primary_ephemeral_pub")
        primary_id_node = reg_node.get_child("primary_identity_pub")
        ref_node = reg_node.get_child("link_code_pairing_ref")

        if (
            not wrapped_node
            or not primary_id_node
            or not ref_node
            or not isinstance(wrapped_node.content, bytes)
            or not isinstance(primary_id_node.content, bytes)
        ):
            raise ValueError("Nodo primary_hello incompleto.")

        wrapped = wrapped_node.content
        salt = wrapped[:32]
        iv = wrapped[32:48]
        ciphertext = wrapped[48:]

        dec_key = _derive_pairing_code_key(self.pairing_code, salt)
        decryptor = Cipher(algorithms.AES(dec_key), modes.CTR(iv)).decryptor()
        primary_ephemeral_pub_bytes = decryptor.update(ciphertext) + decryptor.finalize()

        primary_ephemeral_pub = x25519.X25519PublicKey.from_public_bytes(primary_ephemeral_pub_bytes)
        ephemeral_shared_secret = self._pairing_ephemeral_priv.exchange(primary_ephemeral_pub)

        primary_identity_pub_bytes = primary_id_node.content
        adv_secret_random = os.urandom(32)

        bundle_salt = os.urandom(32)
        bundle_iv = os.urandom(12)
        bundle_enc_key = HKDF(
            algorithm=hashes.SHA256(),
            length=32,
            salt=bundle_salt,
            info=b"link_code_pairing_key_bundle_encryption_key",
        ).derive(ephemeral_shared_secret)

        key_bundle_plain = self.ident_pub_bytes + primary_identity_pub_bytes + adv_secret_random
        encrypted_bundle = AESGCM(bundle_enc_key).encrypt(bundle_iv, key_bundle_plain, b"")
        wrapped_key_bundle = bundle_salt + bundle_iv + encrypted_bundle

        ident_priv = x25519.X25519PrivateKey.from_private_bytes(self.ident_priv_bytes)
        primary_identity_pub = x25519.X25519PublicKey.from_public_bytes(primary_identity_pub_bytes)
        identity_shared_key = ident_priv.exchange(primary_identity_pub)

        adv_ikm = ephemeral_shared_secret + identity_shared_key + adv_secret_random
        self.adv_secret_key = HKDF(
            algorithm=hashes.SHA256(),
            length=32,
            salt=b"",
            info=b"adv_secret",
        ).derive(adv_ikm)

        ref_content = ref_node.content
        if isinstance(ref_content, str):
            ref_content = ref_content.encode("utf-8")

        print("[*] Codigo QR validado... Estableciendo conexión")
        return BinaryNode(
            tag="iq",
            attrs={
                "to": "s.whatsapp.net",
                "type": "set",
                "id": "pairfinish-" + os.urandom(6).hex(),
                "xmlns": "md",
            },
            content=[
                BinaryNode(
                    tag="link_code_companion_reg",
                    attrs={
                        "jid": f"{self.pairing_phone}@s.whatsapp.net",
                        "stage": "companion_finish",
                    },
                    content=[
                        BinaryNode(
                            tag="link_code_pairing_wrapped_key_bundle",
                            content=wrapped_key_bundle,
                        ),
                        BinaryNode(
                            tag="companion_identity_public",
                            content=self.ident_pub_bytes,
                        ),
                        BinaryNode(
                            tag="link_code_pairing_ref",
                            content=ref_content,
                        ),
                    ],
                )
            ],
        )

    def _process_pair_success(self, msg_id: str, pair_success: BinaryNode) -> BinaryNode:
        device_node = pair_success.get_child("device")
        identity_node = pair_success.get_child("device-identity")

        if not device_node or not identity_node or not isinstance(identity_node.content, bytes):
            raise ValueError("Nodo pair-success incompleto")

        self.jid = device_node.attrs.get("jid")
        self.lid = device_node.attrs.get("lid")

        hmac_msg = ADVSignedDeviceIdentityHMAC()
        hmac_msg.ParseFromString(identity_node.content)

        is_hosted = hmac_msg.account_type == HOSTED
        hmac_prefix = b"\x06\x05" if is_hosted else b""
        expected_hmac = hmac.new(
            self.adv_secret_key, hmac_prefix + hmac_msg.details, hashlib.sha256
        ).digest()

        if not hmac.compare_digest(expected_hmac, hmac_msg.hmac):
            raise ValueError("HMAC de device-identity inválido")

        account = ADVSignedDeviceIdentity()
        account.ParseFromString(hmac_msg.details)

        device_prefix = b"\x06\x06" if is_hosted else b"\x06\x01"
        device_msg = (
            device_prefix
            + account.details
            + self.ident_pub_bytes
            + account.account_signature_key
        )
        account.device_signature = xeddsa_sign(self.ident_priv_bytes, device_msg)

        dev_identity = ADVDeviceIdentity()
        dev_identity.ParseFromString(account.details)
        self.key_index = dev_identity.key_index

        self.account_bytes = account.SerializeToString()
        self.save_creds()

        reply_account = ADVSignedDeviceIdentity()
        reply_account.details = account.details
        reply_account.account_signature = account.account_signature
        reply_account.device_signature = account.device_signature
        if is_hosted:
            reply_account.account_signature_key = account.account_signature_key

        print(f"[+] Dispositivo firmado y vinculado: {self.jid} (key-index={self.key_index})")

        return BinaryNode(
            tag="iq",
            attrs={
                "to": "s.whatsapp.net",
                "type": "result",
                "id": msg_id,
            },
            content=[
                BinaryNode(
                    tag="pair-device-sign",
                    content=[
                        BinaryNode(
                            tag="device-identity",
                            attrs={"key-index": str(self.key_index)},
                            content=reply_account.SerializeToString(),
                        )
                    ],
                )
            ],
        )

    def handle_node(self, node: BinaryNode, suppress_qr: bool = False) -> BinaryNode | None:
        if node.tag != "iq" or node.attrs.get("type") != "set":
            return None

        msg_id = node.attrs.get("id", "")

        pair_device = node.get_child("pair-device")
        if pair_device:
            ack = BinaryNode(
                tag="iq",
                attrs={
                    "to": "s.whatsapp.net",
                    "type": "result",
                    "id": msg_id,
                },
            )
            if not suppress_qr:
                refs = pair_device.get_children("ref")
                if refs:
                    ref_raw = refs[0].content
                    ref_str = (
                        ref_raw.decode("utf-8")
                        if isinstance(ref_raw, bytes)
                        else str(ref_raw)
                    )
                    self.print_qr_code(self.build_qr_string(ref_str))
            return ack

        pair_success = node.get_child("pair-success")
        if pair_success:
            print("\n[*] Se ha vinculado el dispositivo con exito.")
            return self._process_pair_success(msg_id, pair_success)

        return None

# Almacén de pre-keys de un solo uso para el protocolo Signal
import base64
import json
import struct
from pathlib import Path

from crypto.keys import generate_x25519_keypair, xeddsa_sign
from binary import BinaryNode

PREKEYS_PATH = Path(__file__).resolve().parent / "prekeys.json"

def _b64e(data: bytes) -> str:
    return base64.b64encode(data).decode("ascii")

def _b64d(text: str) -> bytes:
    return base64.b64decode(text.encode("ascii"))

class PreKeyStore:

    def __init__(self, store_path: Path = PREKEYS_PATH):
        self.store_path = store_path
        self.next_id: int = 1
        self.keys: dict[str, dict[str, str]] = {}

        if self.store_path.exists():
            self._load()

    def _load(self):
        data = json.loads(self.store_path.read_text(encoding="utf-8"))
        self.next_id = int(data.get("next_id", 1))
        self.keys = data.get("keys", {})

    def _save(self):
        self.store_path.parent.mkdir(parents=True, exist_ok=True)
        payload = {
            "next_id": self.next_id,
            "keys": self.keys,
        }
        self.store_path.write_text(json.dumps(payload, indent=2), encoding="utf-8")

    def generate_batch(self, count: int = 30) -> list[tuple[int, bytes]]:
        batch: list[tuple[int, bytes]] = []
        for _ in range(count):
            key_id = self.next_id
            self.next_id += 1
            _, priv_bytes, pub_bytes = generate_x25519_keypair()
            self.keys[str(key_id)] = {
                "priv": _b64e(priv_bytes),
                "pub": _b64e(pub_bytes),
            }
            batch.append((key_id, pub_bytes))
        self._save()
        print(f"[+] {count} Pre-Keys generadas y guardadas en: {self.store_path}")
        return batch

    def get_private_key(self, key_id: int) -> bytes | None:
        entry = self.keys.get(str(key_id))
        return _b64d(entry["priv"]) if entry else None

    def build_upload_iq(
        self,
        iq_id: str,
        reg_id: int,
        ident_priv_bytes: bytes,
        ident_pub_bytes: bytes,
        skey_pub_bytes: bytes,
        count: int = 30,
    ) -> BinaryNode:
        batch = self.generate_batch(count)
        skey_sig = xeddsa_sign(ident_priv_bytes, b"\x05" + skey_pub_bytes)

        key_nodes = [
            BinaryNode(
                tag="key",
                content=[
                    BinaryNode(tag="id", content=struct.pack(">I", key_id)[1:]),
                    BinaryNode(tag="value", content=pub_bytes),
                ],
            )
            for key_id, pub_bytes in batch
        ]

        return BinaryNode(
            tag="iq",
            attrs={
                "id": iq_id,
                "xmlns": "encrypt",
                "type": "set",
                "to": "s.whatsapp.net",
            },
            content=[
                BinaryNode(tag="registration", content=struct.pack(">I", reg_id)),
                BinaryNode(tag="type", content=b"\x05"),
                BinaryNode(tag="identity", content=ident_pub_bytes),
                BinaryNode(tag="list", content=key_nodes),
                BinaryNode(
                    tag="skey",
                    content=[
                        BinaryNode(tag="id", content=struct.pack(">I", 1)[1:]),
                        BinaryNode(tag="value", content=skey_pub_bytes),
                        BinaryNode(tag="signature", content=skey_sig),
                    ],
                ),
            ],
        )

# Utilidad: descarga y regenera binary/tokens.py desde el repositorio oficial de Baileys
import ast
import re
import urllib.request

BAILEYS_CONSTANTS_URL = (
    "https://raw.githubusercontent.com/WhiskeySockets/Baileys/master/src/WABinary/constants.ts"
)

def strip_ts_comments(code: str) -> str:
    pattern = r"('(?:\\.|[^'\\])*'|\"(?:\\.|[^\"\\])*\")|//[^\n]*"
    return re.sub(pattern, lambda m: m.group(1) or "", code)

def extract_array(source: str, var_name: str):
    match = re.search(rf"export const {var_name}\s*=\s*(\[[\s\S]*?\n\])", source)
    if not match:
        raise RuntimeError(f"No se encontró {var_name} en constants.ts")
    raw = strip_ts_comments(match.group(1))
    return ast.literal_eval(raw)

def main():
    print("[*] Descargando diccionario WABinary v3 oficial desde Baileys...")
    req = urllib.request.Request(
        BAILEYS_CONSTANTS_URL,
        headers={"User-Agent": "Mozilla/5.0"},
    )
    with urllib.request.urlopen(req, timeout=15) as resp:
        ts_code = resp.read().decode("utf-8")

    single_tokens = extract_array(ts_code, "SINGLE_BYTE_TOKENS")
    double_tokens = extract_array(ts_code, "DOUBLE_BYTE_TOKENS")

    output_path = "../binary/tokens.py"
    with open(output_path, "w", encoding="utf-8") as f:
        f.write("# Archivo generado automáticamente desde Baileys constants.ts\n\n")
        f.write(f"SINGLE_BYTE_TOKENS = {single_tokens!r}\n\n")
        f.write(f"DOUBLE_BYTE_TOKENS = {double_tokens!r}\n")

    print(
        f"[+] binary/tokens.py generado con éxito "
        f"({len(single_tokens)} tokens simples, {len(double_tokens)} tablas dobles)."
    )

if __name__ == "__main__":
    main()

# wapipeline/delete_msg.py — Eliminar mensajes para todos (REVOKE)
from typing import Any

from protobuf import WAMessage
from .handler import base_jid, is_group, relay_message


def _resolve_delete_target(
    target: Any,
    msg_id: str | None,
    sender: str | None,
    from_me: bool | None,
) -> tuple[str | None, str | None, bool]:
    """Extrae (msg_id, sender_jid, from_me) desde str, dict, IncomingMessage o QuotedMessage."""
    if target is None:
        return msg_id, sender, True if from_me is None else from_me

    if isinstance(target, str):
        return target, sender, True if from_me is None else from_me

    if isinstance(target, dict):
        m_id = target.get("id") or target.get("stanza_id") or msg_id
        m_sender = target.get("sender") or target.get("participant") or sender
        m_from_me = target.get("from_me") if from_me is None else from_me
        return m_id, m_sender, False if m_from_me is None else bool(m_from_me)

    # Objeto estilo QuotedMessage o IncomingMessage
    m_id = getattr(target, "id", None) or getattr(target, "stanza_id", None) or msg_id
    m_sender = getattr(target, "sender", None) or getattr(target, "participant", None) or sender
    obj_from_me = getattr(target, "from_me", None)
    if from_me is not None:
        resolved_from_me = from_me
    elif obj_from_me is not None:
        resolved_from_me = bool(obj_from_me)
    else:
        resolved_from_me = False
    return m_id, m_sender, resolved_from_me


async def delete_msg(
    ws,
    transport,
    signal_session,
    to_jid: str,
    target: Any = None,
    *,
    msg_id: str | None = None,
    sender: str | None = None,
    from_me: bool | None = None,
    debug: bool = False,
) -> str:
    """
    Elimina un mensaje para todos (REVOKE).

    Ejemplos:
        # Eliminar un mensaje enviado por el bot:
        sent_id = await client.send_msg(chat_jid, "Me borro en 3 segundos...")
        await client.delete_msg(chat_jid, sent_id)

        # Eliminar el mensaje citado por el usuario (propio o de otro si el bot es admin):
        if umsg.quoted:
            await client.delete_msg(chat_jid, umsg.quoted)
    """
    m_id, m_sender, is_from_me = _resolve_delete_target(target, msg_id, sender, from_me)
    if not m_id:
        raise ValueError("delete_msg requiere target (msg_id, QuotedMessage o IncomingMessage) o msg_id.")

    group = is_group(to_jid)
    base_to = base_jid(to_jid)

    wa_msg = WAMessage()
    wa_msg.protocol_message.type = 0  # REVOKE
    wa_msg.protocol_message.key.remote_jid = base_to
    wa_msg.protocol_message.key.from_me = is_from_me
    wa_msg.protocol_message.key.id = m_id

    if group and not is_from_me and m_sender:
        wa_msg.protocol_message.key.participant = base_jid(m_sender)

    # edit="7" = borrar mensaje propio | edit="8" = admin borra mensaje de otro en grupo
    edit_code = "7" if is_from_me else ("8" if group else "7")
    participants = [m_sender] if m_sender else None

    revoke_id = await relay_message(
        ws=ws,
        transport=transport,
        signal_session=signal_session,
        to_jid=to_jid,
        wa_msg=wa_msg,
        msg_type="text",
        target_participants=participants,
        extra_attrs={"edit": edit_code},
        debug=debug,
    )
    destino = "GRUPO" if group else "CHAT"
    modo = "PROPIO" if is_from_me else "ADMIN"
    print(f"[>>> ELIMINADO ({destino}/{modo})] target_id={m_id} en {to_jid} (revoke_id={revoke_id})")
    return revoke_id

import time

def _varint(n: int) -> bytes:
    if n < 0:
        n &= (1 << 64) - 1
    out = bytearray()
    while True:
        b = n & 0x7F
        n >>= 7
        if n:
            out.append(b | 0x80)
        else:
            out.append(b)
            break
    return bytes(out)


def _field_varint(field_num: int, val: int) -> bytes:
    return _varint((field_num << 3) | 0) + _varint(val)


def _field_bytes(field_num: int, val: bytes) -> bytes:
    return _varint((field_num << 3) | 2) + _varint(len(val)) + val


def _field_str(field_num: int, val: str) -> bytes:
    return _field_bytes(field_num, val.encode("utf-8"))


def _resolve_edit_msg_id(target: Any, msg_id: str | None) -> str | None:
    if target is None:
        return msg_id
    if isinstance(target, str):
        return target
    if isinstance(target, dict):
        return target.get("id") or target.get("stanza_id") or msg_id
    return getattr(target, "id", None) or getattr(target, "stanza_id", None) or msg_id


def _build_edit_proto(
    remote_jid: str,
    target_msg_id: str,
    edited_wa_msg: WAMessage,
    timestamp_ms: int,
) -> bytes:
    """
    Construye WAMessage con ProtocolMessage (type=14 MESSAGE_EDIT):
      - field 1: key (MessageKey: remote_jid, from_me=True, id)
      - field 2: type = 14 (MESSAGE_EDIT)
      - field 14: edited_message (WAMessage)
      - field 15: timestamp_ms (int64)
    """
    key_bytes = (
        _field_str(1, remote_jid)
        + _field_varint(2, 1)  # from_me = True
        + _field_str(3, target_msg_id)
    )
    proto_msg_bytes = (
        _field_bytes(1, key_bytes)
        + _field_varint(2, 14)  # MESSAGE_EDIT
        + _field_bytes(14, edited_wa_msg.SerializeToString())
        + _field_varint(15, timestamp_ms)
    )
    # WAMessage field 12 = protocol_message
    return _field_bytes(12, proto_msg_bytes)


async def edit_msg(
    ws,
    transport,
    signal_session,
    to_jid: str,
    target: Any,
    new_text: str,
    *,
    msg_id: str | None = None,
    tag: Any = None,
    g_tag: Any = None,
    debug: bool = False,
) -> str:
    """
    Edita un mensaje enviado previamente por el bot.

    Ejemplos:
        sent_id = await client.send_msg(chat_jid, "Texto original")
        await client.edit_msg(chat_jid, sent_id, "Texto editado ✏️")
    """
    target_id = _resolve_edit_msg_id(target, msg_id)
    if not target_id:
        raise ValueError("edit_msg requiere el msg_id del mensaje a editar.")

    base_to = base_jid(to_jid)
    edited_wa = WAMessage()
    target_participants: list[str] | None = None

    if tag or g_tag:
        edited_wa.extended_text_message.text = new_text
        ctx = edited_wa.extended_text_message.context_info
        if tag:
            target_participants = apply_mentions_to_context(ctx, tag)
        if g_tag:
            apply_group_mentions_to_context(ctx, g_tag)
    else:
        edited_wa.conversation = new_text

    raw_proto = _build_edit_proto(
        remote_jid=base_to,
        target_msg_id=target_id,
        edited_wa_msg=edited_wa,
        timestamp_ms=int(time.time() * 1000),
    )

    edit_stanza_id = await relay_message(
        ws=ws,
        transport=transport,
        signal_session=signal_session,
        to_jid=to_jid,
        wa_msg=raw_proto,
        msg_type="text",
        target_participants=target_participants,
        extra_attrs={"edit": "1"},
        debug=debug,
    )
    destino = "GRUPO" if is_group(to_jid) else "CHAT"
    print(f"[>>> EDITADO ({destino})] target_id={target_id} en {to_jid} -> {new_text!r} (id={edit_stanza_id})")
    return edit_stanza_id

def is_group(jid: str) -> bool:
    return "@g.us" in jid


def base_jid(jid: str) -> str:
    user, server = jid.split("@", 1)
    return f"{user.split(':')[0]}@{server}"


def make_msg_id() -> str:
    return "3EB0" + os.urandom(8).hex().upper()


def apply_forward_to_context(
    context_info,
    forward: bool | int = False,
    forwarding_score: int | None = None,
) -> bool:
    """
    Aplica etiqueta de mensaje reenviado (is_forwarded + forwarding_score) en ContextInfo.
- forward=True  -> "Reenviado" (score=1)
- forward=5     -> "Reenviado muchas veces" (score=5)
- forwarding_score=5 -> fija el puntaje explícitamente
    """
    if not forward and forwarding_score is None:
        return False

    if isinstance(forward, bool):
        score = forwarding_score if forwarding_score is not None else (1 if forward else 0)
        is_fwd = forward or (score > 0)
    else:
        score = forwarding_score if forwarding_score is not None else int(forward)
        is_fwd = score > 0

    if not is_fwd:
        return False

    context_info.is_forwarded = True
    context_info.forwarding_score = max(1, int(score))
    return True


def get_contact_devices(signal_session, to_jid: str) -> list[str]:
    """Devuelve todos los dispositivos con sesión activa que pertenecen al mismo usuario."""
    base = base_jid(to_jid)
    base_user, base_server = base.split("@", 1)
    devices = []
    for jid in signal_session.sessions.keys():
        if "@" not in jid:
            continue
        u, s = jid.split("@", 1)
        if s == base_server and u.split(":")[0] == base_user:
            devices.append(jid)
    if to_jid not in devices:
        devices.append(to_jid)
    return devices


async def send_node(ws, transport, node: BinaryNode, debug: bool = False):
    if not ws or not transport:
        raise RuntimeError("No hay conexión activa con WhatsApp.")
    encoded = b"\x00" + encode_binary_node(node)
    encrypted = transport.encrypt(encoded)
    length_prefix = struct.pack(">I", len(encrypted))[1:]
    await ws.send(length_prefix + encrypted)
    if debug:
        print(f"[>>> ENVIADO] {node}")


async def relay_message(
    ws,
    transport,
    signal_session,
    to_jid: str,
    wa_msg: WAMessage | bytes,
    msg_type: str = "text",
    target_participants: list[str] | None = None,
    extra_attrs: dict[str, str] | None = None,
    enc_extra_attrs: dict[str, str] | None = None,
    debug: bool = False,
) -> str:
    """
    Cifra y envía cualquier WAMessage (o bytes proto ya serializados) a un chat individual
    (multi-dispositivo) o a un grupo (skmsg + distribución de SenderKey).
    """
    if not signal_session:
        raise RuntimeError("El motor Signal no está inicializado.")

    raw_proto = wa_msg if isinstance(wa_msg, (bytes, bytearray)) else wa_msg.SerializeToString()
    group = is_group(to_jid)
    base_to = base_jid(to_jid)
    msg_id = make_msg_id()

    if group:
        skmsg_bytes, skdm_bytes = signal_session.encrypt_skmsg(base_to, bytes(raw_proto))

        skdm_wa = WAMessage()
        skdm_wa.sender_key_distribution_message.group_id = base_to
        skdm_wa.sender_key_distribution_message.axolotl_sender_key_distribution_message = skdm_bytes
        skdm_proto = skdm_wa.SerializeToString()

        enc_msg_attrs = {"v": "2", "type": "msg"}
        if msg_type == "reaction":
            enc_msg_attrs["decrypt-fail"] = "hide"

        participant_nodes = []
        jids_to_notify = set(target_participants or []) | set(signal_session.sessions.keys())
        uses_lid = any(j.endswith("@lid") for j in jids_to_notify)

        for p_jid in jids_to_notify:
            if p_jid in signal_session.sessions:
                try:
                    enc_skdm = signal_session.encrypt_msg(p_jid, skdm_proto)
                    participant_nodes.append(
                        BinaryNode(
                            tag="to",
                            attrs={"jid": p_jid},
                            content=[
                                BinaryNode(tag="enc", attrs=dict(enc_msg_attrs), content=enc_skdm)
                            ],
                        )
                    )
                except Exception:
                    pass

        children = []
        if msg_type == "reaction":
            children.append(BinaryNode(tag="meta", attrs={"content_type": "add_on"}))
        if participant_nodes:
            children.append(BinaryNode(tag="participants", content=participant_nodes))

        skmsg_attrs = {"v": "2", "type": "skmsg"}
        if msg_type == "reaction":
            skmsg_attrs["decrypt-fail"] = "hide"
        if enc_extra_attrs:
            skmsg_attrs.update(enc_extra_attrs)

        children.append(BinaryNode(tag="enc", attrs=skmsg_attrs, content=skmsg_bytes))

        msg_attrs = {"id": msg_id, "to": base_to, "type": msg_type}
        if uses_lid:
            msg_attrs["addressing_mode"] = "lid"
        if extra_attrs:
            msg_attrs.update(extra_attrs)

        node = BinaryNode(
            tag="message",
            attrs=msg_attrs,
            content=children,
        )
    else:
        devices = get_contact_devices(signal_session, to_jid)
        participant_nodes = []
        enc_attrs = {"v": "2", "type": "msg"}
        if msg_type == "reaction":
            enc_attrs["decrypt-fail"] = "hide"
        if enc_extra_attrs:
            enc_attrs.update(enc_extra_attrs)

        for dev_jid in devices:
            try:
                enc_bytes = signal_session.encrypt_msg(dev_jid, bytes(raw_proto))
                participant_nodes.append(
                    BinaryNode(
                        tag="to",
                        attrs={"jid": dev_jid},
                        content=[
                            BinaryNode(tag="enc", attrs=dict(enc_attrs), content=enc_bytes)
                        ],
                    )
                )
            except Exception as e:
                print(f"[!] No se pudo cifrar ({msg_type}) para {dev_jid}: {e}")

        if not participant_nodes:
            raise RuntimeError(f"No hay sesión Signal activa para {to_jid}")

        msg_attrs = {"id": msg_id, "to": base_to, "type": msg_type}
        if extra_attrs:
            msg_attrs.update(extra_attrs)

        node = BinaryNode(
            tag="message",
            attrs=msg_attrs,
            content=[BinaryNode(tag="participants", content=participant_nodes)],
        )

    await send_node(ws, transport, node, debug)
    return msg_id

def _split_jid(jid: str | None) -> tuple[str | None, str | None, str | None]:
    """Devuelve (user_base, device, server) a partir de un JID."""
    if not jid or "@" not in jid:
        return None, None, None
    user_part, server = jid.split("@", 1)
    if ":" in user_part:
        user_base, dev = user_part.split(":", 1)
        return user_base, dev, server
    return user_part, None, server


def _base_jid(jid: str | None) -> str | None:
    user_base, _, server = _split_jid(jid)
    if not user_base or not server:
        return jid
    return f"{user_base}@{server}"

async def keepalive_loop(ws, transport, debug: bool = False):
    seq = 1
    try:
        while True:
            await asyncio.sleep(25)
            ping = BinaryNode(
                tag="iq",
                attrs={
                    "id": f"ping-{seq}",
                    "to": "s.whatsapp.net",
                    "type": "get",
                    "xmlns": "w:p",
                },
                content=[BinaryNode(tag="ping")],
            )
            await send_node(ws, transport, ping, debug)
            if debug:
                print(f"[*] Keep-Alive ping-{seq} enviado.")
            seq += 1
    except asyncio.CancelledError:
        pass

async def handle_node(node: BinaryNode, client) -> bool:
    if node.tag == "success":
        return False

    if node.tag == "iq" and node.attrs.get("type") in ("result", "error"):
        iq_id = node.attrs.get("id")
        pending = getattr(client, "_pending_iqs", None)
        if iq_id and pending and iq_id in pending:
            fut = pending[iq_id]
            if not fut.done():
                fut.set_result(node)
            return False

    if node.tag == "ack" and node.attrs.get("class") == "message":
        print(f"[ACK] {node}")
        return False

    if node.tag == "stream:error" and node.attrs.get("code") == "515":
        print("[*] Reinicio de stream (515). Reconectando...")
        return True

    if node.tag in ("notification", "receipt", "call"):
        if node.tag == "receipt":
            print(f"[RECIBED] {node}")
        await send_node(client._ws, client._transport, build_ack(node))
        if node.tag == "notification":
            notif_type = node.attrs.get("type")
            if notif_type == "encrypt":
                await _handle_prekey_refill(node, client)
            elif notif_type == "link_code_companion_reg" or node.get_child("link_code_companion_reg"):
                finish_iq = client.pairing.handle_link_code_primary_hello(node)
                if finish_iq is not None:
                    await send_node(client._ws, client._transport, finish_iq, client.debug)
        return False

    if node.tag == "message":
        await _handle_message(node, client)
        return False

    use_pair_code = bool(getattr(client, "phone_number", None)) and not client.pairing.is_registered
    is_pair_device = node.tag == "iq" and node.attrs.get("type") == "set" and node.get_child("pair-device") is not None

    reply = client.pairing.handle_node(node, suppress_qr=use_pair_code)
    if reply is not None:
        await send_node(client._ws, client._transport, reply, client.debug)

    if is_pair_device and use_pair_code and not getattr(client, "_pairing_code_requested", False):
        client._pairing_code_requested = True
        asyncio.create_task(client.request_pairing_code(client.phone_number))

    return False


async def _handle_prekey_refill(node: BinaryNode, client):
    count_node = node.get_child("count")
    if count_node and int(count_node.attrs.get("value", "0")) < 10:
        client._prekey_seq = getattr(client, "_prekey_seq", 1)
        iq_id = f"upload-prekeys-{client._prekey_seq}"
        client._prekey_seq += 1
        upload_iq = client.prekeys.build_upload_iq(
            iq_id=iq_id,
            reg_id=client.pairing.reg_id,
            ident_priv_bytes=client.pairing.ident_priv_bytes,
            ident_pub_bytes=client.pairing.ident_pub_bytes,
            skey_pub_bytes=client.pairing.skey_pub_bytes,
            count=30,
        )
        await send_node(client._ws, client._transport, upload_iq, client.debug)


def _try_extract_skdm(plain: bytes, group_jid: str, sender: str, signal) -> bool:
    try:
        wa = WAMessage()
        wa.ParseFromString(plain)
        if wa.HasField("sender_key_distribution_message"):
            skdm_inner = wa.sender_key_distribution_message.axolotl_sender_key_distribution_message
            if skdm_inner:
                signal.process_sender_key_distribution(group_jid, sender, skdm_inner)
                return True
    except Exception:
        pass
    return False


def _log_reaction_if_present(plain: bytes, sender: str):

        wa = WAMessage()
        wa.ParseFromString(plain)
        if wa.HasField("react_message"):
            r = wa.react_message
            print(
                f"[REACT: {sender}] "
                f"emoji={r.text!r} remote_jid={r.key.remote_jid!r} "
                f"participant={r.key.participant!r} from_me={r.key.from_me} "
                f"target_id={r.key.id!r} ts={r.sender_timestamp_ms}"
            )
    except Exception:
        pass


async def _handle_message(node: BinaryNode, client):
    chat_jid = node.attrs.get("from", "")
    sender = node.attrs.get("participant") or chat_jid
    is_grp = "@g.us" in chat_jid
    my_jids = [client.pairing.jid, client.pairing.lid]

    enc = node.get_child("enc")
    text = None
    mentioned_jids: list[str] = []
    g_mentions: list[dict[str, str]] = []
    quoted: QuotedMessage | None = None
    is_forwarded = False
    forwarding_score = 0
    decrypt_failed = False

    if enc and isinstance(enc.content, bytes):
        enc_type = enc.attrs.get("type")

        try:
            if enc_type == "pkmsg":
                plain = client.signal.decrypt_pkmsg(sender, enc.content)
                if is_grp:
                    _try_extract_skdm(plain, chat_jid, sender, client.signal)
                _log_reaction_if_present(plain, sender)
                text = extract_text(plain)
                mentioned_jids = extract_mentions(plain)
                g_mentions = extract_group_mentions(plain)
                quoted = extract_quoted(plain, chat_jid, my_jids=my_jids)
                is_forwarded, forwarding_score = extract_forward_info(plain)

            elif enc_type == "msg":
                plain = client.signal.decrypt_msg(sender, enc.content)
                if is_grp:
                    _try_extract_skdm(plain, chat_jid, sender, client.signal)
                _log_reaction_if_present(plain, sender)
                text = extract_text(plain)
                mentioned_jids = extract_mentions(plain)
                g_mentions = extract_group_mentions(plain)
                quoted = extract_quoted(plain, chat_jid, my_jids=my_jids)
                is_forwarded, forwarding_score = extract_forward_info(plain)

            elif enc_type == "skmsg":
                try:
                    plain = client.signal.decrypt_skmsg(chat_jid, sender, enc.content)
                    _log_reaction_if_present(plain, sender)
                    text = extract_text(plain)
                    mentioned_jids = extract_mentions(plain)
                    g_mentions = extract_group_mentions(plain)
                    quoted = extract_quoted(plain, chat_jid, my_jids=my_jids)
                    is_forwarded, forwarding_score = extract_forward_info(plain)
                except ValueError as ve:
                    if "No hay sender key" in str(ve):
                        print(f"[!] Sender key ausente para {sender} en {chat_jid}. Pidiendo retry...")
                        decrypt_failed = True
                    else:
                        raise
            else:
                print(f"[!] Tipo de enc desconocido: {enc_type!r} de {sender}")

        except Exception as e:
            decrypt_failed = True
            print(f"[!] Error descifrando ({enc_type}) de {sender}: {e}")

    await send_node(client._ws, client._transport, build_ack(node))

    if decrypt_failed:
        print(f"[*] Solicitando reenvío (retry + keys) a {sender}...")
        await send_node(
            client._ws, client._transport,
            build_retry_receipt(node, client.pairing, client.prekeys),
        )
        return

    if text is not None:
        _print_incoming(node, sender, chat_jid, text)
        sender_pn = node.attrs.get("sender_pn") or node.attrs.get("participant_pn")
        umsg = IncomingMessage(
            id=node.attrs.get("id", ""),
            sender=sender,
            chat_jid=chat_jid,
            sender_pn=sender_pn,
            push_name=node.attrs.get("notify"),
            text=text,
            addressing_mode=node.attrs.get("addressing_mode"),
            mentioned_jids=mentioned_jids,
            g_mentions=g_mentions,
            quoted=quoted,
            is_forwarded=is_forwarded,
            forwarding_score=forwarding_score,
            raw_node=node,
        )
        await client._emit("message", umsg)

def _b64url(data: bytes) -> str:
    return base64.urlsafe_b64encode(data).decode("ascii").rstrip("=")


def _download_url_sync(url: str, timeout: float = 20.0) -> bytes:
    req = urllib.request.Request(url, headers={"User-Agent": MEDIA_USER_AGENT})
    with urllib.request.urlopen(req, timeout=timeout) as resp:
        return resp.read()


async def load_media_bytes(source: str | Path | bytes, timeout: float = 20.0) -> bytes:
    """Carga bytes desde una URL (http/https), una ruta local o bytes directamente."""
    if isinstance(source, (bytes, bytearray)):
        return bytes(source)

    if isinstance(source, str):
        stripped = source.strip()
        if stripped.startswith(("http://", "https://")):
            return await asyncio.to_thread(_download_url_sync, stripped, timeout)
        path = Path(stripped)
    else:
        path = source

    if not path.exists():
        raise FileNotFoundError(f"No se encontró el archivo: {path}")
    return await asyncio.to_thread(path.read_bytes)


async def send_iq_and_wait(
    ws,
    transport,
    pending_iqs: dict[str, asyncio.Future],
    iq_node: BinaryNode,
    debug: bool = False,
    timeout: float = 15.0,
) -> BinaryNode:
    iq_id = iq_node.attrs["id"]
    loop = asyncio.get_running_loop()
    fut: asyncio.Future = loop.create_future()
    pending_iqs[iq_id] = fut
    try:
        await send_node(ws, transport, iq_node, debug)
        return await asyncio.wait_for(fut, timeout=timeout)
    finally:
        pending_iqs.pop(iq_id, None)


async def request_media_conn(
    ws,
    transport,
    pending_iqs: dict[str, asyncio.Future],
    cache: dict | None = None,
    debug: bool = False,
) -> dict:
    """Obtiene (o reutiliza si no expiró) el auth y los hosts de subida media_conn."""
    now = time.time()
    if cache and cache.get("expires_at", 0) > now + 30:
        return cache

    iq_node = BinaryNode(
        tag="iq",
        attrs={
            "id": "mconn-" + os.urandom(6).hex(),
            "xmlns": "w:m",
            "type": "set",
            "to": "s.whatsapp.net",
        },
        content=[BinaryNode(tag="media_conn")],
    )
    resp = await send_iq_and_wait(ws, transport, pending_iqs, iq_node, debug=debug)
    if resp.attrs.get("type") == "error":
        raise RuntimeError(f"WhatsApp rechazó media_conn: {resp}")

    mc = resp.get_child("media_conn")
    if not mc:
        raise RuntimeError(f"Respuesta sin nodo <media_conn>: {resp}")

    auth = mc.attrs.get("auth", "")
    ttl = int(mc.attrs.get("ttl", "300"))
    hosts = [h.attrs["hostname"] for h in mc.get_children("host") if h.attrs.get("hostname")]
    if not auth or not hosts:
        raise RuntimeError(f"media_conn incompleto: {mc}")

    result = {
        "auth": auth,
        "hosts": hosts,
        "expires_at": now + ttl,
    }
    if cache is not None:
        cache.clear()
        cache.update(result)
    return result


def prepare_sticker_webp(source: str | Path | bytes) -> tuple[bytes, int, int]:
    """
    Convierte una imagen (bytes o ruta local) a WebP 512x512 si no es WebP.
    Retorna (webp_bytes, width, height).
    """
    if isinstance(source, (str, Path)):
        path = Path(source)
        if not path.exists():
            raise FileNotFoundError(f"No se encontró el archivo de sticker: {path}")
        raw = path.read_bytes()
    else:
        raw = bytes(source)

    # Si ya es WebP, devolver directo
    if raw[:4] == b"RIFF" and raw[8:12] == b"WEBP":
        return raw, 512, 512

    try:
        from PIL import Image
    except ImportError as e:
        raise RuntimeError(
            "Se necesita Pillow para convertir PNG/JPG a sticker WebP: py -m pip install Pillow"
        ) from e

    with Image.open(io.BytesIO(raw)) as img:
        img = img.convert("RGBA")
        img.thumbnail((512, 512), Image.Resampling.LANCZOS)
        canvas = Image.new("RGBA", (512, 512), (0, 0, 0, 0))
        offset = ((512 - img.width) // 2, (512 - img.height) // 2)
        canvas.paste(img, offset, img)
        out = io.BytesIO()
        canvas.save(out, format="WEBP", quality=80, method=6)
        return out.getvalue(), 512, 512


def encrypt_media(data: bytes, media_type: str = "sticker") -> dict:
    """Cifra un archivo con el esquema estándar de WhatsApp (HKDF + AES-256-CBC + HMAC-SHA256[:10])."""
    info = MEDIA_HKDF_INFO.get(media_type, b"WhatsApp Image Keys")
    media_key = os.urandom(32)
    expanded = HKDF(
        algorithm=hashes.SHA256(),
        length=112,
        salt=b"\x00" * 32,
        info=info,
    ).derive(media_key)

    iv = expanded[0:16]
    cipher_key = expanded[16:48]
    mac_key = expanded[48:80]

    padder = sym_padding.PKCS7(128).padder()
    padded = padder.update(data) + padder.finalize()
    encryptor = Cipher(algorithms.AES(cipher_key), modes.CBC(iv)).encryptor()
    ciphertext = encryptor.update(padded) + encryptor.finalize()

    mac = hmac.new(mac_key, iv + ciphertext, hashlib.sha256).digest()[:10]
    enc_data = ciphertext + mac

    return {
        "media_key": media_key,
        "media_key_timestamp": int(time.time()),
        "file_sha256": hashlib.sha256(data).digest(),
        "file_enc_sha256": hashlib.sha256(enc_data).digest(),
        "file_length": len(data),
        "enc_data": enc_data,
    }


def _http_upload_sync(
    hosts: list[str],
    auth: str,
    media_type: str,
    enc_data: bytes,
    file_enc_sha256: bytes,
) -> tuple[str, str]:
    path_prefix = MEDIA_PATH_MAP.get(media_type, "/mms/image")
    token = _b64url(file_enc_sha256)
    auth_q = urllib.parse.quote(auth, safe="")
    last_err: Exception | None = None

    for host in hosts:
        url = f"https://{host}{path_prefix}/{token}?auth={auth_q}&token={token}"
        req = urllib.request.Request(
            url,
            data=enc_data,
            method="POST",
            headers={
                "Content-Type": "application/octet-stream",
                "Origin": "https://web.whatsapp.com",
                "Referer": "https://web.whatsapp.com/",
            },
        )
        try:
            with urllib.request.urlopen(req, timeout=25) as resp:
                body = json.loads(resp.read().decode("utf-8"))
                media_url = body.get("url")
                direct_path = body.get("direct_path")
                if media_url and direct_path:
                    return media_url, direct_path
        except Exception as e:
            last_err = e
            continue

    raise RuntimeError(f"Error subiendo media a WhatsApp MMG: {last_err}")


async def upload_media_file(
    ws,
    transport,
    pending_iqs: dict[str, asyncio.Future],
    data: bytes,
    media_type: str = "sticker",
    media_conn_cache: dict | None = None,
    debug: bool = False,
) -> dict:
    """Cifra y sube un archivo a los servidores MMG de WhatsApp."""
    mc = await request_media_conn(ws, transport, pending_iqs, cache=media_conn_cache, debug=debug)
    enc = encrypt_media(data, media_type=media_type)
    media_url, direct_path = await asyncio.to_thread(
        _http_upload_sync,
        mc["hosts"],
        mc["auth"],
        media_type,
        enc["enc_data"],
        enc["file_enc_sha256"],
    )
    enc["url"] = media_url
    enc["direct_path"] = direct_path
    return enc

# wapipeline/react_msg.py — Reacciones con emoji a mensajes
import time

def _resolve_target(
    target: Any,
    target_msg_id: str | None,
    target_sender: str | None,
) -> tuple[str | None, str | None]:
    """Extrae (msg_id, sender_jid) desde IncomingMessage, dict, str o campos sueltos."""
    if target is None:
        return target_msg_id, target_sender

    if isinstance(target, str):
        return target, target_sender

    if isinstance(target, dict):
        m_id = target.get("id") or target.get("stanza_id") or target_msg_id
        m_sender = target.get("sender") or target.get("participant") or target_sender
        return m_id, m_sender

    # Objeto estilo IncomingMessage
    m_id = getattr(target, "id", None) or getattr(target, "stanza_id", None) or target_msg_id
    m_sender = getattr(target, "sender", None) or getattr(target, "participant", None) or target_sender
    return m_id, m_sender


async def react_msg(
    ws,
    transport,
    signal_session,
    to_jid: str,
    emoji: str,
    target: Any = None,
    *,
    target_msg_id: str | None = None,
    target_sender: str | None = None,
    debug: bool = False,
) -> str:
    """
    Reacciona a un mensaje con un emoji.

    Ejemplos:
        await react_msg(..., chat_jid, "👍", target=msg)
        await react_msg(..., chat_jid, "👍", target=msg.id, target_sender=msg.sender)
    """
    m_id, m_sender = _resolve_target(target, target_msg_id, target_sender)
    if not m_id:
        raise ValueError("react_msg requiere target (IncomingMessage) o target_msg_id.")

    group = is_group(to_jid)
    base_to = base_jid(to_jid)

    wa_msg = WAMessage()
    wa_msg.react_message.key.remote_jid = base_to
    wa_msg.react_message.key.from_me = False
    wa_msg.react_message.key.id = m_id
    if group and m_sender:
        wa_msg.react_message.key.participant = base_jid(m_sender)
    wa_msg.react_message.text = emoji
    wa_msg.react_message.sender_timestamp_ms = int(time.time() * 1000)

    participants = [m_sender] if m_sender else None

    msg_id = await relay_message(
        ws=ws,
        transport=transport,
        signal_session=signal_session,
        to_jid=to_jid,
        wa_msg=wa_msg,
        msg_type="reaction",
        target_participants=participants,
        debug=debug,
    )
    destino = "GRUPO" if group else "CHAT"
    print(f"[>>> REACCIÓN '{emoji}' ({destino})] sobre msg_id={m_id} en {to_jid} (id={msg_id})")
    return msg_id

import time

async def _send_single_read(
    ws,
    transport,
    msg_id: str,
    sender_jid: str,
    chat_jid: str | None = None,
    debug: bool = False,
) -> None:
    to_jid = chat_jid if chat_jid else sender_jid

    attrs: dict[str, str] = {
        "id": msg_id,
        "to": base_jid(to_jid),

        "t": t_now,
    }
    if group:
        attrs["participant"] = base_jid(sender_jid)

    node = BinaryNode(tag="receipt", attrs=attrs)
    await send_node(ws, transport, node, debug)
    destino = "GRUPO" if group else "CHAT"
    print(f"[>>> LEÍDO ({destino})] msg_id={msg_id} sender={sender_jid} t={t_now}")


async def read_msg(
    ws,
    transport,
    target: Any = None,
    sender_jid: str | None = None,
    chat_jid: str | None = None,
    debug: bool = False,
    *,
    msg_id: str | None = None,
) -> None:
    """
    Marca uno o varios mensajes como leídos (doble tick azul).

    Ejemplos:
        await read_msg(..., msg)                # IncomingMessage
        await read_msg(..., [msg1, msg2])       # lista de IncomingMessage o dicts
        await read_msg(..., msg_id="...", sender_jid="...", chat_jid="...")
    """
    if isinstance(target, (list, tuple)):
        for item in target:
            await read_msg(ws, transport, item, sender_jid=sender_jid, chat_jid=chat_jid, debug=debug)
        return

    if target is not None and not isinstance(target, str):
        # IncomingMessage o dict
        if isinstance(target, dict):
            resolved_id = target.get("id") or msg_id
            resolved_sender = target.get("sender") or sender_jid
            resolved_chat = target.get("chat_jid") if target.get("chat_jid") != resolved_sender else chat_jid
        else:
            resolved_id = getattr(target, "id", None) or msg_id
            resolved_sender = getattr(target, "sender", None) or sender_jid
            is_grp = getattr(target, "is_group", False)
            raw_chat = getattr(target, "chat_jid", None)
            resolved_chat = (raw_chat if is_grp else None) if chat_jid is None else chat_jid
    else:
        resolved_id = msg_id or target
        resolved_sender = sender_jid
        resolved_chat = chat_jid

    if not resolved_id or not resolved_sender:
        raise ValueError("read_msg requiere msg_id/IncomingMessage y sender_jid.")

    await _send_single_read(
        ws=ws,
        transport=transport,
        msg_id=resolved_id,
        sender_jid=resolved_sender,
        chat_jid=resolved_chat,
        debug=debug,
    )

# Procesamiento de mensajes entrantes: extracción de texto, menciones, quoted, forward, ACKs y retry
import struct
import time

from protobuf import WAMessage
from binary import BinaryNode
from crypto.keys import xeddsa_sign


def _base_jid(jid: str | None) -> str | None:
    """Devuelve el JID base (sin device) a partir de un JID completo."""
    if not jid or "@" not in jid:
        return jid
    user_part, server = jid.split("@", 1)
    base_user = user_part.split(":")[0]
    return f"{base_user}@{server}"


def _extract_text_from_wa_msg(msg: WAMessage) -> str | None:
    if msg.HasField("conversation"):
        return msg.conversation
    if msg.HasField("extended_text_message") and msg.extended_text_message.HasField("text"):
        return msg.extended_text_message.text
    return None


def extract_text(raw_proto: bytes) -> str | None:
    msg = WAMessage()
    msg.ParseFromString(raw_proto)
    return _extract_text_from_wa_msg(msg)


def extract_mentions(raw_proto: bytes) -> list[str]:
    msg = WAMessage()
    msg.ParseFromString(raw_proto)
    if (
        msg.HasField("extended_text_message")
        and msg.extended_text_message.HasField("context_info")
    ):
        return list(msg.extended_text_message.context_info.mentioned_jid)
    return []


def extract_group_mentions(raw_proto: bytes) -> list[dict[str, str]]:
    msg = WAMessage()
    msg.ParseFromString(raw_proto)
    if (
        msg.HasField("extended_text_message")
        and msg.extended_text_message.HasField("context_info")
    ):
        return [
            {
                "group_jid": gm.group_jid,
                "group_subject": gm.group_subject,
            }
            for gm in msg.extended_text_message.context_info.group_mentions
            if gm.group_jid
        ]
    return []


def extract_forward_info(raw_proto: bytes) -> tuple[bool, int]:
    """Devuelve (is_forwarded, forwarding_score) desde ContextInfo."""
    msg = WAMessage()
    msg.ParseFromString(raw_proto)
    if (
        msg.HasField("extended_text_message")
        and msg.extended_text_message.HasField("context_info")
    ):
        ctx = msg.extended_text_message.context_info
        return bool(ctx.is_forwarded), int(ctx.forwarding_score)
    return False, 0


def extract_quoted(
    raw_proto: bytes,
    chat_jid: str,
    my_jids: list[str | None] | None = None,
):
    """Extrae el mensaje citado (reply) desde ContextInfo si existe."""
    from .handler import QuotedMessage  # deferred to avoid circular import at load time

    msg = WAMessage()
    msg.ParseFromString(raw_proto)
    if not (
        msg.HasField("extended_text_message")
        and msg.extended_text_message.HasField("context_info")
    ):
        return None

    ctx = msg.extended_text_message.context_info
    if not ctx.stanza_id:
        return None

    q_sender = ctx.participant or chat_jid
    q_text = _extract_text_from_wa_msg(ctx.quoted_message) if ctx.HasField("quoted_message") else None

    my_base_set = {_base_jid(j) for j in (my_jids or []) if j}
    q_sender_base = _base_jid(q_sender)
    is_from_me = bool(q_sender_base and q_sender_base in my_base_set)

    return QuotedMessage(
        id=ctx.stanza_id,
        sender=q_sender,
        chat_jid=chat_jid,
        text=q_text,
        from_me=is_from_me,
    )


def print_incoming_message(node: BinaryNode, sender: str, text: str):
    push_name = node.attrs.get("notify", "")
    msg_id = node.attrs.get("id", "")
    label = f"{push_name} ({sender})" if push_name else sender
    print(f"[<<< MENSAJE de {label}] {text!r}  (id={msg_id})")


def build_ack(node: BinaryNode) -> BinaryNode:
    attrs = {
        "id": node.attrs.get("id", ""),
        "to": node.attrs.get("from", "s.whatsapp.net"),
        "class": node.tag,
    }
    if node.tag != "message" and "type" in node.attrs:
        attrs["type"] = node.attrs["type"]
    if "participant" in node.attrs:
        attrs["participant"] = node.attrs["participant"]
    return BinaryNode(tag="ack", attrs=attrs)


def build_retry_receipt(node: BinaryNode, pairing, prekeys) -> BinaryNode:
    msg_id = node.attrs.get("id", "")
    ts = node.attrs.get("t", str(int(time.time())))
    attrs = {"id": msg_id, "to": node.attrs.get("from", ""), "type": "retry", "t": ts}
    if "participant" in node.attrs:
        attrs["participant"] = node.attrs["participant"]

    key_id, key_pub = prekeys.generate_batch(1)[0]
    skey_sig = xeddsa_sign(pairing.ident_priv_bytes, b"\x05" + pairing.skey_pub_bytes)

    keys_children = [
        BinaryNode(tag="type", content=b"\x05"),
        BinaryNode(tag="identity", content=pairing.ident_pub_bytes),
        BinaryNode(tag="key", content=[
            BinaryNode(tag="id", content=struct.pack(">I", key_id)[1:]),
            BinaryNode(tag="value", content=key_pub),
        ]),
        BinaryNode(tag="skey", content=[
            BinaryNode(tag="id", content=struct.pack(">I", 1)[1:]),
            BinaryNode(tag="value", content=pairing.skey_pub_bytes),
            BinaryNode(tag="signature", content=skey_sig),
        ]),
    ]
    if pairing.account_bytes:
        keys_children.append(BinaryNode(tag="device-identity", content=pairing.account_bytes))

    return BinaryNode(
        tag="receipt",
        attrs=attrs,
        content=[
            BinaryNode(tag="retry", attrs={"v": "1", "count": "2", "id": msg_id, "t": ts}),
            BinaryNode(tag="registration", content=struct.pack(">I", pairing.reg_id)),
            BinaryNode(tag="keys", content=keys_children),
        ],
    )

# wapipeline/send_msg.py — Envío de mensajes de texto, reply, menciones de usuario y grupo

def _resolve_single_jid(user: Any) -> str | None:
    """Convierte un IncomingMessage, dict, número o JID a su JID base limpio."""
    if user is None:
        return None

    if isinstance(user, str):
        raw = user.strip().lstrip("@")
        if not raw:
            return None
        if "@" in raw:
            return base_jid(raw)
        digits = "".join(ch for ch in raw if ch.isdigit())
        return f"{digits}@s.whatsapp.net" if digits else None

    if isinstance(user, dict):
        raw = user.get("sender") or user.get("jid") or user.get("participant") or user.get("number")
        return _resolve_single_jid(raw)

    # Objeto estilo IncomingMessage
    raw = getattr(user, "sender_base", None) or getattr(user, "sender", None) or getattr(user, "jid", None)
    return _resolve_single_jid(raw)


def normalize_mention_jids(users: Any) -> list[str]:
    """Normaliza uno o varios usuarios a una lista sin duplicados de JIDs base."""
    if users is None:
        return []
    items = users if isinstance(users, (list, tuple, set)) else [users]
    result: list[str] = []
    seen: set[str] = set()
    for item in items:
        jid = _resolve_single_jid(item)
        if jid and jid not in seen:
            seen.add(jid)
            result.append(jid)
    return result


def tag_user(user: Any) -> str:
    """Devuelve el tag visible '@<id>' que coincide con el JID en mentioned_jid."""
    jid = _resolve_single_jid(user)
    if not jid:
        return ""
    return f"@{jid.split('@', 1)[0]}"


def apply_mentions_to_context(context_info, mentions: Any) -> list[str]:
    """Inyecta los JIDs normalizados en ContextInfo.mentioned_jid y los devuelve."""
    jids = normalize_mention_jids(mentions)
    if jids:
        context_info.mentioned_jid.extend(jids)
    return jids

def _normalize_group_jid(raw_jid: str | None) -> str | None:
    """Convierte un JID o ID de grupo a formato '<id>@g.us' limpio."""
    if not raw_jid or not isinstance(raw_jid, str):
        return None
    cleaned = raw_jid.strip().lstrip("@")
    if not cleaned:
        return None
    if "@" in cleaned:
        resolved = base_jid(cleaned)
        return resolved if resolved.endswith("@g.us") else None
    return f"{cleaned}@g.us"


def _resolve_single_group(
    group: Any,
    default_subject: str | None = None,
) -> tuple[str, str] | None:
    """Extrae (group_jid, group_subject) desde str, tuple, dict o IncomingMessage."""
    if group is None:
        return None

    fallback_subject = (default_subject or "Grupo").strip() or "Grupo"

    if isinstance(group, str):
        g_jid = _normalize_group_jid(group)
        return (g_jid, fallback_subject) if g_jid else None

    if isinstance(group, tuple) and len(group) == 2 and isinstance(group[0], str):
        g_jid = _normalize_group_jid(group[0])
        g_sub = str(group[1]).strip() if group[1] else fallback_subject
        return (g_jid, g_sub or fallback_subject) if g_jid else None

    if isinstance(group, dict):
        raw_jid = (
            group.get("group_jid")
            or group.get("groupJid")
            or group.get("jid")
            or group.get("chat_jid")
        )
        raw_sub = (
            group.get("group_subject")
            or group.get("groupSubject")
            or group.get("subject")
            or group.get("name")
            or fallback_subject
        )
        g_jid = _normalize_group_jid(raw_jid)
        return (g_jid, str(raw_sub).strip() or fallback_subject) if g_jid else None

    # Objeto estilo IncomingMessage o GroupMetadata
    raw_jid = getattr(group, "chat_jid", None) or getattr(group, "jid", None)
    raw_sub = getattr(group, "subject", None) or fallback_subject
    g_jid = _normalize_group_jid(raw_jid)
    return (g_jid, str(raw_sub).strip() or fallback_subject) if g_jid else None


def normalize_group_mentions(
    groups: Any,
    default_subject: str | None = None,
) -> list[tuple[str, str]]:
    """
    Normaliza uno o varios grupos a una lista sin duplicados de (group_jid, group_subject).
    """
    if groups is None:
        return []

    if (
        isinstance(groups, tuple)
        and len(groups) == 2
        and isinstance(groups[0], str)
        and isinstance(groups[1], str)
        and "@g.us" not in groups[1]
    ):
        items = [groups]
    elif isinstance(groups, (list, tuple, set)):
        items = list(groups)
    else:
        items = [groups]

    result: list[tuple[str, str]] = []
    seen: set[str] = set()
    for item in items:
        resolved = _resolve_single_group(item, default_subject=default_subject)
        if resolved and resolved[0] not in seen:
            seen.add(resolved[0])
            result.append(resolved)
    return result


def tag_group(group: Any) -> str:
    """Devuelve el token '@<id>@g.us' que WhatsApp sustituye por '@<group_subject>'."""
    resolved = _resolve_single_group(group)
    if not resolved:
        return ""
    return f"@{resolved[0]}"


def apply_group_mentions_to_context(
    context_info,
    g_tag: Any,
    default_subject: str | None = None,
) -> list[tuple[str, str]]:
    """Inyecta los GroupMention normalizados en ContextInfo.group_mentions y los devuelve."""
    items = normalize_group_mentions(g_tag, default_subject=default_subject)
    for g_jid, g_subject in items:
        gm = context_info.group_mentions.add()
        gm.group_jid = g_jid
        gm.group_subject = g_subject
    return items

def _resolve_quote(
    quoted: Any,
    quoted_msg_id: str | None,
    quoted_sender: str | None,
    quoted_text: str | None,
) -> tuple[str | None, str | None, str]:
    """Extrae (stanza_id, participant, quoted_text) desde IncomingMessage, dict, str o campos sueltos."""
    if quoted is None:
        return quoted_msg_id, quoted_sender, quoted_text or ""

    if isinstance(quoted, str):
        return quoted_msg_id, quoted_sender, quoted

    if isinstance(quoted, dict):
        q_id = quoted.get("id") or quoted.get("stanza_id") or quoted_msg_id
        q_sender = quoted.get("sender") or quoted.get("participant") or quoted_sender
        q_text = quoted.get("text") or quoted.get("conversation") or quoted_text or ""
        return q_id, q_sender, q_text

    # Objeto estilo IncomingMessage / QuotedMessage
    q_id = getattr(quoted, "id", None) or getattr(quoted, "stanza_id", None) or quoted_msg_id
    q_sender = getattr(quoted, "sender", None) or getattr(quoted, "participant", None) or quoted_sender
    q_text = getattr(quoted, "text", None) or getattr(quoted, "conversation", None) or quoted_text or ""
    return q_id, q_sender, q_text

async def send_msg(
    ws,
    transport,
    signal_session,
    to_jid: str,
    text: str,
    debug: bool = False,
    *,
    reply: Any = None,
    quoted_msg_id: str | None = None,
    quoted_sender: str | None = None,
    quoted_text: str | None = None,
    tag: Any = None,
    g_tag: Any = None,
    forward: bool | int = False,
    forwarding_score: int | None = None,
) -> str:
    """
    Envía texto a un chat individual o grupo.

    Parámetros opcionales:
      - reply: IncomingMessage, QuotedMessage, dict o str con el mensaje a citar.
               También puedes pasar quoted_msg_id / quoted_sender / quoted_text sueltos.
      - tag: uno o varios usuarios a mencionar (@usuario).
      - g_tag: uno o varios grupos a mencionar (@Grupo).
      - forward / forwarding_score: etiqueta de reenviado.
    """
    wa_msg = WAMessage()
    target_participants: list[str] = []

    has_reply = bool(reply is not None or quoted_msg_id or quoted_sender or quoted_text)
    has_context = bool(has_reply or tag or g_tag or forward or forwarding_score is not None)

    if has_context:
        wa_msg.extended_text_message.text = text
        ctx = wa_msg.extended_text_message.context_info

        if has_reply:
            q_id, q_sender, q_text = _resolve_quote(reply, quoted_msg_id, quoted_sender, quoted_text)
            if q_id:
                ctx.stanza_id = q_id
            if q_sender:
                ctx.participant = base_jid(q_sender)
                target_participants.append(q_sender)
            if q_text:
                ctx.quoted_message.conversation = q_text

        if tag:
            target_participants.extend(apply_mentions_to_context(ctx, tag))
        if g_tag:
            apply_group_mentions_to_context(ctx, g_tag)
        apply_forward_to_context(ctx, forward=forward, forwarding_score=forwarding_score)
    else:
        wa_msg.conversation = text

    msg_id = await relay_message(
        ws=ws,
        transport=transport,
        signal_session=signal_session,
        to_jid=to_jid,
        wa_msg=wa_msg,
        msg_type="text",
        target_participants=target_participants or None,
        debug=debug,
    )
    destino = "GRUPO" if is_group(to_jid) else "CHAT"
    if has_reply:
        ref_label = (
            getattr(reply, "id", None)
            or quoted_msg_id
            or "quote"
        )
        print(f"[>>> REPLY ({destino}) a {to_jid} citando {ref_label}] {text!r} (id={msg_id})")
    else:
        print(f"[>>> TEXTO ENVIADO ({destino}) a {to_jid}] {text!r} (id={msg_id})")
    return msg_id

async def mention_user(
    ws,
    transport,
    signal_session,
    to_jid: str,
    users: Any,
    text: str = "",
    *,
    ghost: bool = False,
    reply: Any = None,
    quoted_msg_id: str | None = None,
    quoted_sender: str | None = None,
    quoted_text: str | None = None,
    forward: bool | int = False,
    forwarding_score: int | None = None,
    debug: bool = False,
) -> str:
    """
    Menciona a uno o varios usuarios en un chat o grupo.

    - ghost=False (por defecto): si el texto no contiene los '@<id>', los antepone automáticamente.
    - ghost=True: mención oculta/fantasma (notifica al usuario sin mostrar el '@' en el texto).
    - reply: opcionalmente cita un mensaje al mismo tiempo.
    - forward / forwarding_score: etiqueta de reenviado opcional.
    """
    mentioned_jids = normalize_mention_jids(users)
    if not mentioned_jids:
        raise ValueError("mention_user requiere al menos un usuario válido para mencionar.")

    final_text = text
    if not ghost:
        missing_tags = [
            f"@{jid.split('@', 1)[0]}"
            for jid in mentioned_jids
            if f"@{jid.split('@', 1)[0]}" not in final_text
        ]
        if missing_tags:
            prefix = " ".join(missing_tags)
            final_text = f"{prefix} {final_text}".strip() if final_text else prefix

    wa_msg = WAMessage()
    wa_msg.extended_text_message.text = final_text
    ctx = wa_msg.extended_text_message.context_info
    ctx.mentioned_jid.extend(mentioned_jids)
    apply_forward_to_context(ctx, forward=forward, forwarding_score=forwarding_score)

    target_participants = list(mentioned_jids)

    if reply is not None or quoted_msg_id or quoted_sender or quoted_text:
        q_id, q_sender, q_text = _resolve_quote(reply, quoted_msg_id, quoted_sender, quoted_text)
        if q_id:
            ctx.stanza_id = q_id
        if q_sender:
            ctx.participant = base_jid(q_sender)
            target_participants.append(q_sender)
        if q_text:
            ctx.quoted_message.conversation = q_text

    msg_id = await relay_message(
        ws=ws,
        transport=transport,
        signal_session=signal_session,
        to_jid=to_jid,
        wa_msg=wa_msg,
        msg_type="text",
        target_participants=target_participants,
        debug=debug,
    )
    destino = "GRUPO" if is_group(to_jid) else "CHAT"
    modo = "GHOST" if ghost else "MENTION"
    print(f"[>>> {modo} ({destino}) a {to_jid} -> {mentioned_jids}] {final_text!r} (id={msg_id})")
    return msg_id

import asyncio

async def send_sticker(
    ws,
    transport,
    signal_session,
    pending_iqs: dict[str, asyncio.Future],
    to_jid: str,
    sticker: str | Path | bytes,
    *,
    quoted: Any = None,
    quoted_msg_id: str | None = None,
    quoted_sender: str | None = None,
    quoted_text: str | None = None,
    tag: Any = None,
    g_tag: Any = None,
    forward: bool | int = False,
    forwarding_score: int | None = None,
    is_animated: bool = False,
    media_conn_cache: dict | None = None,
    debug: bool = False,
) -> str:
    """
    Envía un sticker normal desde un archivo local, una URL (http/https) o bytes.

    Ejemplos:
        await client.send_sticker(chat_jid, "src/prueba.png")
        await client.send_sticker(chat_jid, "https://ejemplo.com/imagen.png", quoted=umsg)
    """
    raw_bytes = await load_media_bytes(sticker)
    webp_bytes, width, height = await asyncio.to_thread(prepare_sticker_webp, raw_bytes)

    uploaded = await upload_media_file(
        ws=ws,
        transport=transport,
        pending_iqs=pending_iqs,
        data=webp_bytes,
        media_type="sticker",
        media_conn_cache=media_conn_cache,
        debug=debug,
    )

    context_info_bytes: bytes | None = None
    target_participants: list[str] = []

    has_quote = bool(quoted is not None or quoted_msg_id or quoted_sender or quoted_text)
    has_ctx = bool(has_quote or tag or g_tag or forward or forwarding_score is not None)
    if has_ctx:
        tmp_wa = WAMessage()
        ctx = tmp_wa.extended_text_message.context_info
        if has_quote:
            q_id, q_sender, q_text = _resolve_quote(quoted, quoted_msg_id, quoted_sender, quoted_text)
            if q_id:
                ctx.stanza_id = q_id
            if q_sender:
                ctx.participant = base_jid(q_sender)
                target_participants.append(q_sender)
            if q_text:
                ctx.quoted_message.conversation = q_text
        if tag:
            target_participants.extend(apply_mentions_to_context(ctx, tag))
        if g_tag:
            apply_group_mentions_to_context(ctx, g_tag)
        apply_forward_to_context(ctx, forward=forward, forwarding_score=forwarding_score)
        context_info_bytes = ctx.SerializeToString()

    raw_proto = build_sticker_proto(
        url=uploaded["url"],
        direct_path=uploaded["direct_path"],
        media_key=uploaded["media_key"],
        file_sha256=uploaded["file_sha256"],
        file_enc_sha256=uploaded["file_enc_sha256"],
        file_length=uploaded["file_length"],
        media_key_timestamp=uploaded["media_key_timestamp"],
        width=width,
        height=height,
        is_animated=is_animated,
        context_info_bytes=context_info_bytes,
        sticker_sent_ts=int(time.time() * 1000),
    )

    msg_id = await relay_message(
        ws=ws,
        transport=transport,
        signal_session=signal_session,
        to_jid=to_jid,
        wa_msg=raw_proto,
        msg_type="media",
        target_participants=target_participants or None,
        enc_extra_attrs={"mediatype": "sticker"},
        debug=debug,
    )
    destino = "GRUPO" if is_group(to_jid) else "CHAT"
    print(f"[>>> STICKER ENVIADO ({destino}) a {to_jid}] (id={msg_id})")
    return msg_id

def _resolve_target(
    target: Any,
    target_msg_id: str | None,
    target_sender: str | None,
    from_me: bool | None,
) -> tuple[str | None, str | None, bool]:
    """Extrae (msg_id, sender_jid, from_me) desde IncomingMessage, QuotedMessage, dict o str."""
    if target is None:
        return target_msg_id, target_sender, bool(from_me) if from_me is not None else False

    if isinstance(target, str):
        return target, target_sender, bool(from_me) if from_me is not None else False

    if isinstance(target, dict):
        m_id = target.get("id") or target.get("stanza_id") or target_msg_id
        m_sender = target.get("sender") or target.get("participant") or target_sender
        m_from_me = target.get("from_me") if from_me is None else from_me
        return m_id, m_sender, bool(m_from_me) if m_from_me is not None else False

    m_id = getattr(target, "id", None) or getattr(target, "stanza_id", None) or target_msg_id
    m_sender = getattr(target, "sender", None) or getattr(target, "participant", None) or target_sender
    obj_from_me = getattr(target, "from_me", None)
    resolved_from_me = bool(from_me) if from_me is not None else bool(obj_from_me)
    return m_id, m_sender, resolved_from_me


async def sticker_react(
    ws,
    transport,
    signal_session,
    pending_iqs: dict[str, asyncio.Future],
    to_jid: str,
    target: Any,
    sticker: str | Path | bytes,
    *,
    target_msg_id: str | None = None,
    target_sender: str | None = None,
    from_me: bool | None = None,
    is_animated: bool = False,
    media_conn_cache: dict | None = None,
    debug: bool = False,
) -> str:
    """
    Reacciona a un mensaje con un sticker (Sticker Annotation).
    Acepta archivo local, URL (http/https) o bytes.
    """
    m_id, m_sender, is_from_me = _resolve_target(target, target_msg_id, target_sender, from_me)
    if not m_id:
        raise ValueError("sticker_react requiere target (IncomingMessage/QuotedMessage/msg_id).")

    raw_bytes = await load_media_bytes(sticker)
    webp_bytes, width, height = await asyncio.to_thread(prepare_sticker_webp, raw_bytes)
    uploaded = await upload_media_file(
        ws=ws,
        transport=transport,
        pending_iqs=pending_iqs,
        data=webp_bytes,
        media_type="sticker",
        media_conn_cache=media_conn_cache,
        debug=debug,
    )

    group = is_group(to_jid)
    base_to = base_jid(to_jid)
    parent_participant = base_jid(m_sender) if (group and not is_from_me and m_sender) else None

    raw_proto = build_sticker_reaction_proto(
        url=uploaded["url"],
        direct_path=uploaded["direct_path"],
        media_key=uploaded["media_key"],
        file_sha256=uploaded["file_sha256"],
        file_enc_sha256=uploaded["file_enc_sha256"],
        file_length=uploaded["file_length"],
        media_key_timestamp=uploaded["media_key_timestamp"],
        parent_remote_jid=base_to,
        parent_msg_id=m_id,
        parent_from_me=is_from_me,
        parent_participant=parent_participant,
        width=width,
        height=height,
        is_animated=is_animated,
        sticker_sent_ts=int(time.time() * 1000),
    )

    participants = [m_sender] if m_sender else None

    msg_id = await relay_message(
        ws=ws,
        transport=transport,
        signal_session=signal_session,
        to_jid=to_jid,
        wa_msg=raw_proto,
        msg_type="media",
        target_participants=participants,
        enc_extra_attrs={"mediatype": "sticker"},
        debug=debug,
    )
    destino = "GRUPO" if group else "CHAT"
    print(f"[>>> STICKER REACT ({destino})] sobre msg_id={m_id} en {to_jid} (id={msg_id})")
    return msg_id

# wapipeline/typing.py — Todos los estados de chatstate y presencia
# Soporta: composing (escribiendo), recording (grabando audio), paused, available (online), unavailable (offline)
from binary import BinaryNode
from .handler import base_jid, is_group, send_node

_STATE_ALIASES = {
    "composing": "composing",
    "typing": "composing",
    "recording": "recording",
    "audio": "recording",
    "paused": "paused",
    "stop": "paused",
    "available": "available",
    "online": "available",
    "unavailable": "unavailable",
    "offline": "unavailable",
}


async def send_presence(
    ws,
    transport,
    state: str,
    to_jid: str | None = None,
    debug: bool = False,
) -> None:
    """
    Envía cualquier estado de presencia o chatstate de WhatsApp:
      - "composing" / "typing"    → "escribiendo..." (requiere to_jid)
      - "recording" / "audio"     → "grabando audio..." (requiere to_jid)
      - "paused" / "stop"         → detiene escribiendo/grabando (requiere to_jid)
      - "available" / "online"    → en línea (presencia global)
      - "unavailable" / "offline" → desconectado (presencia global)
    """
    normalized = _STATE_ALIASES.get(state.strip().lower())
    if not normalized:
        valid = ", ".join(sorted(set(_STATE_ALIASES.values())))
        raise ValueError(f"Estado inválido {state!r}. Usa uno de: {valid}")

    if normalized in ("available", "unavailable"):
        node = BinaryNode(tag="presence", attrs={"type": normalized})
        await send_node(ws, transport, node, debug)
        label = "EN LÍNEA" if normalized == "available" else "DESCONECTADO"
        print(f"[>>> PRESENCE] {label} ({normalized})")
        return

    if not to_jid:
        raise ValueError(f"El estado '{normalized}' requiere un to_jid.")

    target = base_jid(to_jid)
    group = is_group(target)

    if normalized == "composing":
        child = BinaryNode(tag="composing")
        estado = "ESCRIBIENDO"
    elif normalized == "recording":
        child = BinaryNode(tag="composing", attrs={"media": "audio"})
        estado = "GRABANDO AUDIO"
    else:
        child = BinaryNode(tag="paused")
        estado = "PAUSADO"

    node = BinaryNode(
        tag="chatstate",
        attrs={"to": target},
        content=[child],
    )
    await send_node(ws, transport, node, debug)
    destino = "GRUPO" if group else "CHAT"
    print(f"[>>> CHATSTATE ({destino})] {estado} → {to_jid}")


async def send_typing(
    ws,
    transport,
    to_jid: str,
    composing: bool = True,
    debug: bool = False,
) -> None:
    """Muestra u oculta el indicador 'escribiendo...' en la conversación."""
    await send_presence(ws, transport, "composing" if composing else "paused", to_jid, debug)


async def stop_typing(ws, transport, to_jid: str, debug: bool = False) -> None:
    """Atajo para cancelar el indicador de escritura."""
    await send_presence(ws, transport, "paused", to_jid, debug)

    to_jid: str,

    await send_presence(ws, transport, "paused", to_jid, debug)


async def set_online(ws, transport, debug: bool = False) -> None:
    """Marca la cuenta como en línea (available)."""
    await send_presence(ws, transport, "available", debug=debug)

    await send_presence(ws, transport, "unavailable", debug=debug)
